/**
 * 1.7 bg-n10 <b>轮7</b> A 级探针（崩乱合金：配方 / 块属性 / 信标 / §9.7 豁免 / 两个状态公式 /
 * 50% 挂失衡 / 区间乒乓 / 重锤 ×1.4 / 盾牌 / 盔甲 / 禁果）。
 *
 * <p>临时探针（收尾时整块删除）。开关文件：{@code run/bgn10r7-probe.enabled}。</p>
 *
 * <p>？ 探针纪律：{@code halt(false)} 只调一次；forceload 双向归零；绝不进生产代码路径。</p>
 */
public final class BgN10r7Probe {

    private static final String TAG = "BGN10R7PROBE";

    private static int passes;
    private static int fails;
    private static int ticks;
    private static boolean done;

    private static final List<int[]> FORCED = new ArrayList<>();

    private static int baseX;
    private static int baseY;
    private static int baseZ;

    private static ServerLevel level;
    private static ServerPlayer player;

    private BgN10r7Probe() {
    }

    private static Path gameDir() {
        return Path.of(System.getProperty("user.dir"));
    }

    public static boolean enabled() {
        return Files.exists(gameDir().resolve("bgn10r7-probe.enabled"));
    }

    public static void install() {
        if (!enabled()) {
            return;
        }
        bettergold.LOGGER.info("{} probe registered (switch file found) gameDir={}", TAG, gameDir());
        NeoForge.EVENT_BUS.register(BgN10r7Probe.class);
    }

    private static void pass(String name, String detail) {
        passes++;
        bettergold.LOGGER.info("{} PASS {} :: {}", TAG, name, detail);
    }

    private static void fail(String name, String detail) {
        fails++;
        bettergold.LOGGER.error("{} FAIL {} :: {}", TAG, name, detail);
    }

    private static void check(String name, boolean ok, String detail) {
        if (ok) {
            pass(name, detail);
        } else {
            fail(name, detail);
        }
    }

    private static String num(double v) {
        return String.format(Locale.ROOT, "%.4f", v);
    }

    private static ServerPlayer newPlayer(MinecraftServer server, ServerLevel level, String name) {
        ServerPlayer p = new ServerPlayer(server, level, new GameProfile(UUID.randomUUID(), name),
                net.minecraft.server.level.ClientInformation.createDefault());
        // ⚠ 必须有 connection：`ItemCooldowns#addCooldown` 会 `player.connection.send(...)`，
        //   connection 为 null 时抛 NPE（轮7 首跑实测：禁果与盾牌两组各炸一次）。
        //   形状与轮6 探针逐字同构。
        p.connection = new net.minecraft.server.network.ServerGamePacketListenerImpl(
                server,
                new net.minecraft.network.Connection(net.minecraft.network.protocol.PacketFlow.SERVERBOUND),
                p,
                net.minecraft.server.network.CommonListenerCookie.createInitial(
                        new GameProfile(UUID.randomUUID(), "bgn10r7probe"), false));
        p.setGameMode(GameType.SURVIVAL);
        return p;
    }

    // ==================== 生命周期 ====================

    @SubscribeEvent
    public static void onTick(ServerTickEvent.Post event) {
        if (done) {
            return;
        }
        ticks++;
        if (ticks == 4) {
            setup(event.getServer());
        } else if (ticks == 12) {
            try {
                runAll();
            } catch (Throwable t) {
                fail("r7_uncaught", t.getClass().getSimpleName() + ": " + t.getMessage());
            }
            summaryAndHalt(event.getServer());
        }
    }

    private static void setup(MinecraftServer server) {
        level = server.overworld();
        BlockPos spawn = level.getSharedSpawnPos();
        baseX = spawn.getX() + 40;
        baseY = spawn.getY() + 2;
        baseZ = spawn.getZ() + 40;
        for (int dx = -1; dx <= 1; dx++) {
            for (int dz = -1; dz <= 1; dz++) {
                int cx = (baseX >> 4) + dx;
                int cz = (baseZ >> 4) + dz;
                level.setChunkForced(cx, cz, true);
                FORCED.add(new int[] { cx, cz });
            }
        }
        // 平台：保证脚下有实体地面
        for (int dx = -2; dx <= 2; dx++) {
            for (int dz = -2; dz <= 2; dz++) {
                level.setBlockAndUpdate(new BlockPos(baseX + dx, baseY - 1, baseZ + dz),
                        Blocks.STONE.defaultBlockState());
            }
        }
        player = newPlayer(server, level, "bgn10r7probe");
        // 新造的 ServerPlayer 带出生无敌时间 ⇒ 所有 hurt 都会被拒（首跑实测 hurtApplied=false）⇒ 反射清零
        clearSpawnInvulnerable(player);
        player.moveTo(baseX + 0.5D, baseY, baseZ + 0.5D, 0.0F, 0.0F);
        check("r7_forceload_set", level.getForcedChunks().size() >= 9,
                "forcedAfter=" + level.getForcedChunks().size());
        String worldName = "-";
        try {
            // 客户端 LevelData 没有名字访问器（AGENTS 红线 5）⇒ 从整合服务端的 ServerLevelData 反射读
            Object data = level.getLevelData();
            worldName = String.valueOf(data.getClass().getMethod("getLevelName").invoke(data));
        } catch (Throwable ignored) {
            // 读不到就留 "-"，不影响任何判定
        }
        bettergold.LOGGER.info("{} INFO world :: name={} base=({},{},{})", TAG, worldName, baseX, baseY, baseZ);
    }

    private static void summaryAndHalt(MinecraftServer server) {
        done = true;
        for (int[] c : FORCED) {
            level.setChunkForced(c[0], c[1], false);
        }
        FORCED.clear();
        bettergold.LOGGER.info("{} SUMMARY pass={} fail={} forcedNow={}", TAG, passes, fails,
                level.getForcedChunks().size());
        server.halt(false);
    }

    private static Zombie zombie(double dx, double dz) {
        Zombie z = new Zombie(EntityType.ZOMBIE, level);
        z.moveTo(baseX + dx, baseY, baseZ + dz, 0.0F, 0.0F);
        z.setNoAi(true);
        z.setPersistenceRequired();
        z.getAttribute(Attributes.ARMOR).setBaseValue(0.0D);
        z.getAttribute(Attributes.ARMOR_TOUGHNESS).setBaseValue(0.0D);
        z.setHealth(z.getMaxHealth());
        level.addFreshEntity(z);
        return z;
    }

    private static ItemStack stack(String id) {
        var item = net.minecraft.core.registries.BuiltInRegistries.ITEM
                .get(ResourceLocation.fromNamespaceAndPath("bettergold", id));
        if (item == null || item == Items.AIR) {
            throw new IllegalStateException("未注册物品 bettergold:" + id);
        }
        return new ItemStack(item);
    }

    private static float lost(float before, float after) {
        return before - after;
    }

    // ---------- ① 配方：真配方表匹配（不是读 JSON） ----------
    private static void recipes() {
        try {
            ItemStack ingot = stack("chaosalloy_ingot");
            List<ItemStack> grid = new ArrayList<>();
            for (int i = 0; i < 4; i++) {
                grid.add(new ItemStack(Items.NETHERITE_INGOT));
            }
            for (int i = 0; i < 4; i++) {
                grid.add(stack("chaosgold_block"));
            }
            grid.add(stack("refined_alchemic_oil"));
            CraftingInput in = CraftingInput.of(3, 3, grid);
            ItemStack out = level.getRecipeManager().getRecipeFor(RecipeType.CRAFTING, in, level)
                    .map(h -> h.value().getResultItem(level.registryAccess())).orElse(ItemStack.EMPTY);
            check("r7_recipe_ingot", out.getItem() == ingot.getItem() && out.getCount() == 1,
                    "out=" + out.getItem() + " count=" + out.getCount());

            // 模板：3×3 逐格
            List<ItemStack> t = new ArrayList<>();
            String[] rows = { "GIG", "GAG", "GCG" };
            for (String row : rows) {
                for (char c : row.toCharArray()) {
                    t.add(switch (c) {
                        case 'G' -> stack("chaosgold_ingot");
                        case 'I' -> new ItemStack(Items.NETHERITE_UPGRADE_SMITHING_TEMPLATE);
                        case 'A' -> stack("chaosalloy_ingot");
                        default -> stack("chaosgold_block");
                    });
                }
            }
            ItemStack tplOut = level.getRecipeManager()
                    .getRecipeFor(RecipeType.CRAFTING, CraftingInput.of(3, 3, t), level)
                    .map(h -> h.value().getResultItem(level.registryAccess())).orElse(ItemStack.EMPTY);
            check("r7_recipe_template", tplOut.getItem() == stack("chaosalloy_upgrade_template").getItem(),
                    "out=" + tplOut.getItem());

            // 块：9 锭
            List<ItemStack> b = new ArrayList<>();
            for (int i = 0; i < 9; i++) {
                b.add(stack("chaosalloy_ingot"));
            }
            ItemStack bOut = level.getRecipeManager()
                    .getRecipeFor(RecipeType.CRAFTING, CraftingInput.of(3, 3, b), level)
                    .map(h -> h.value().getResultItem(level.registryAccess())).orElse(ItemStack.EMPTY);
            check("r7_recipe_block", bOut.getItem() == stack("chaosalloy_block").getItem(),
                    "out=" + bOut.getItem());

            // 禁果：混沌禁果 + 崩乱合金锭（无序）
            List<ItemStack> f = List.of(stack("chaosgold_forbidden_fruit"), stack("chaosalloy_ingot"));
            ItemStack fOut = level.getRecipeManager()
                    .getRecipeFor(RecipeType.CRAFTING, CraftingInput.of(3, 3, pad(f)), level)
                    .map(h -> h.value().getResultItem(level.registryAccess())).orElse(ItemStack.EMPTY);
            check("r7_recipe_fruit", fOut.getItem() == stack("chaosalloy_forbidden_fruit").getItem(),
                    "out=" + fOut.getItem());
        } catch (Throwable t) {
            fail("r7_recipes", t.toString());
        }

        // 15 条锻造：逐件走真配方表
        String[] gear = { "sword", "mace", "trident", "bow", "crossbow", "axe", "pickaxe", "shovel",
                "hoe", "shield", "helmet", "chestplate", "leggings", "boots", "knife" };
        List<String> bad = new ArrayList<>();
        int ok = 0;
        for (String g : gear) {
            try {
                ItemStack tpl = stack("chaosalloy_upgrade_template");
                ItemStack base = stack("chaosgold_" + g);
                ItemStack add = stack("chaosalloy_ingot");
                Optional<? extends net.minecraft.world.item.crafting.RecipeHolder<?>> hit = level
                        .getRecipeManager().getRecipeFor(RecipeType.SMITHING,
                                new SmithingRecipeInput(tpl, base, add), level);
                if (hit.isEmpty()) {
                    bad.add(g + ":nomatch");
                    continue;
                }
                ItemStack out = ((net.minecraft.world.item.crafting.Recipe<?>) hit.get().value())
                        .getResultItem(level.registryAccess());
                if (out.getItem() == stack("chaosalloy_" + g).getItem()) {
                    ok++;
                } else {
                    bad.add(g + ":" + out.getItem());
                }
            } catch (Throwable t) {
                bad.add(g + ":" + t.getClass().getSimpleName());
            }
        }
        check("r7_recipe_smithing_all", bad.isEmpty() && ok == gear.length,
                "ok=" + ok + "/" + gear.length + " bad=" + bad);
    }

    private static List<ItemStack> pad(List<ItemStack> in) {
        List<ItemStack> out = new ArrayList<>(in);
        while (out.size() < 9) {
            out.add(ItemStack.EMPTY);
        }
        return out;
    }

    // ---------- ② 块：硬度 / 抗爆 ×3、信标基座、§9.7 豁免 ----------
    private static void blockAndExempt() {
        try {
            BlockState ca = stack("chaosalloy_block").getItem() instanceof net.minecraft.world.item.BlockItem bi
                    ? bi.getBlock().defaultBlockState()
                    : null;
            if (ca == null) {
                fail("r7_block_hardness", "不是方块物品");
                return;
            }
            BlockState netherite = Blocks.NETHERITE_BLOCK.defaultBlockState();
            float hCa = ca.getDestroySpeed(level, new BlockPos(baseX, baseY, baseZ));
            float hNe = netherite.getDestroySpeed(level, new BlockPos(baseX, baseY, baseZ));
            float rCa = ca.getBlock().getExplosionResistance();
            float rNe = netherite.getBlock().getExplosionResistance();
            check("r7_block_hardness_x3", Math.abs(hCa - 150.0F) < 1e-4 && Math.abs(hCa - 3.0F * hNe) < 1e-4,
                    "chaosalloy=" + num(hCa) + " netherite=" + num(hNe) + " ratio=" + num(hCa / hNe));
            check("r7_block_resistance_x3", Math.abs(rCa - 3600.0F) < 1e-4 && Math.abs(rCa - 3.0F * rNe) < 1e-4,
                    "chaosalloy=" + num(rCa) + " netherite=" + num(rNe) + " ratio=" + num(rCa / rNe));
            check("r7_block_beacon_base", ca.is(BlockTags.BEACON_BASE_BLOCKS),
                    "tag=" + ca.is(BlockTags.BEACON_BASE_BLOCKS)
                            + " netheriteControl=" + netherite.is(BlockTags.BEACON_BASE_BLOCKS));

            // §9.7：例外集（privé static，反射读；正负对照）
            var m = Class.forName("com.hjmmd_8.bettergold.event.ModEvents")
                    .getDeclaredMethod("isDustConversionExempt", BlockState.class);
            m.setAccessible(true);
            boolean exemptCa = (Boolean) m.invoke(null, ca);
            boolean exemptNe = (Boolean) m.invoke(null, netherite);
            boolean exemptStone = (Boolean) m.invoke(null, Blocks.STONE.defaultBlockState());
            check("r7_dust_exempt_positive", exemptCa,
                    "chaosalloy_block exempt=" + exemptCa + "（家族索引自动覆盖，零代码）");
            check("r7_dust_exempt_negative", exemptNe && !exemptStone,
                    "netherite_block exempt=" + exemptNe + " stone exempt=" + exemptStone);

            // 真事件：古董镐挖崩乱合金块 ⇒ 掉落**不被就地转换**（仍是崩乱合金块）
            List<ItemEntity> drops = new ArrayList<>();
            drops.add(new ItemEntity(level, baseX + 0.5D, baseY + 0.5D, baseZ + 0.5D, stack("chaosalloy_block")));
            BlockDropsEvent ev = new BlockDropsEvent(level, new BlockPos(baseX, baseY, baseZ), ca, null, drops,
                    player, stack("antique_pickaxe"));
            net.neoforged.neoforge.common.NeoForge.EVENT_BUS.post(ev);
            String after = ev.getDrops().isEmpty() ? "empty" : ev.getDrops().get(0).getItem().getItem().toString();
            check("r7_dust_event_antique_keeps_block",
                    !ev.getDrops().isEmpty()
                            && ev.getDrops().get(0).getItem().getItem() == stack("chaosalloy_block").getItem(),
                    "dropsAfter=" + after + "（走的是 §9.7 的 if (!conversionExempt) 守卫）");
        } catch (Throwable t) {
            fail("r7_block", t.toString());
        }
    }

    // ---------- ③ 两个状态：公式 + 无等级 ----------
    private static void effectsAndFormulas() {
        try {
            // 无等级：施加后 amplifier 恒 0、时长 320
            Zombie z = zombie(3.0D, 0.0D);
            ImbalanceEffects.apply(z, AllEffects.POWER_IMBALANCE.getDelegate());
            MobEffectInstance inst = z.getEffect(AllEffects.POWER_IMBALANCE);
            check("r7_effect_no_level_power", inst != null && inst.getAmplifier() == 0
                            && inst.getDuration() == ImbalanceEffects.TICKS,
                    "amplifier=" + (inst == null ? "-" : inst.getAmplifier())
                            + " duration=" + (inst == null ? "-" : inst.getDuration()));
            ImbalanceEffects.apply(z, AllEffects.RESISTANCE_IMBALANCE.getDelegate());
            MobEffectInstance inst2 = z.getEffect(AllEffects.RESISTANCE_IMBALANCE);
            check("r7_effect_no_level_resistance", inst2 != null && inst2.getAmplifier() == 0
                            && inst2.getDuration() == ImbalanceEffects.TICKS,
                    "amplifier=" + (inst2 == null ? "-" : inst2.getAmplifier())
                            + " duration=" + (inst2 == null ? "-" : inst2.getDuration()));

            // 抗性失衡：受到 4 点 ⇒ 8 点（固定 ×2）
            Zombie victim = zombie(6.0D, 0.0D);
            victim.removeAllEffects();
            float before = victim.getHealth();
            victim.hurt(level.damageSources().generic(), 4.0F);
            float control = lost(before, victim.getHealth());
            victim.setHealth(victim.getMaxHealth());
            victim.invulnerableTime = 0;
            ImbalanceEffects.apply(victim, AllEffects.RESISTANCE_IMBALANCE.getDelegate());
            before = victim.getHealth();
            victim.hurt(level.damageSources().generic(), 4.0F);
            float doubled = lost(before, victim.getHealth());
            check("r7_resistance_imbalance_x2", Math.abs(control - 4.0F) < 1e-4 && Math.abs(doubled - 8.0F) < 1e-4,
                    "control=" + num(control) + " withEffect=" + num(doubled) + "（固定 ×2）");

            // 力量失衡：攻击者在 1 / 原本 之间两端乒乓
            Zombie attacker = zombie(9.0D, 0.0D);
            Zombie target = zombie(9.0D, 2.0D);
            target.removeAllEffects();
            List<String> seq = new ArrayList<>();
            for (int i = 0; i < 4; i++) {
                target.setHealth(target.getMaxHealth());
                target.invulnerableTime = 0;
                float b = target.getHealth();
                target.hurt(level.damageSources().mobAttack(attacker), 6.0F);
                seq.add(num(lost(b, target.getHealth())));
            }
            // 对照：不挂效果 ⇒ 每次都是原本的 6
            List<String> ctl = new ArrayList<>();
            for (int i = 0; i < 2; i++) {
                target.setHealth(target.getMaxHealth());
                target.invulnerableTime = 0;
                float b = target.getHealth();
                target.hurt(level.damageSources().mobAttack(attacker), 6.0F);
                ctl.add(num(lost(b, target.getHealth())));
            }
            check("r7_power_imbalance_control", ctl.equals(List.of("6.0000", "6.0000")),
                    "control=" + ctl);
            ImbalanceEffects.apply(attacker, AllEffects.POWER_IMBALANCE.getDelegate());
            for (int i = 0; i < 4; i++) {
                target.setHealth(target.getMaxHealth());
                target.invulnerableTime = 0;
                float b = target.getHealth();
                target.hurt(level.damageSources().mobAttack(attacker), 6.0F);
                seq.add(num(lost(b, target.getHealth())));
            }
            List<String> tail = seq.subList(4, 8);
            boolean onlyEndpoints = tail.stream().allMatch(s -> s.equals("1.0000") || s.equals("6.0000"));
            boolean bothSeen = tail.contains("1.0000") && tail.contains("6.0000");
            check("r7_power_imbalance_endpoints", onlyEndpoints && bothSeen,
                    "controlThenEffect=" + tail + "（最低 1 / 最高原本 6，两端乒乓）");
        } catch (Throwable t) {
            fail("r7_effects", t.toString());
        }
    }

    // ---------- ④ 50% 挂失衡：近战 / 远程 / 盔甲 ----------
    private static void imbalanceChance() {
        try {
            MetalFamily fam = AllMetals.CHAOSALLOY;
            Zombie target = zombie(-3.0D, 0.0D);
            ItemStack sword = stack("chaosalloy_sword");
            int n = 400;
            int hits = 0;
            int power = 0;
            int resist = 0;
            for (int i = 0; i < n; i++) {
                target.removeAllEffects();
                MetalEvents.dispatchWeaponHit(player, sword, target);
                if (target.hasEffect(AllEffects.POWER_IMBALANCE)) {
                    hits++;
                    power++;
                } else if (target.hasEffect(AllEffects.RESISTANCE_IMBALANCE)) {
                    hits++;
                    resist++;
                }
            }
            double ratio = hits / (double) n;
            check("r7_imbalance_melee_50pct", ratio > 0.42 && ratio < 0.58 && power > 0 && resist > 0,
                    "n=" + n + " hits=" + hits + " ratio=" + num(ratio)
                            + " power=" + power + " resistance=" + resist
                            + "（50% 挂不挂 × 50/50 二选一）");

            // 远程：真箭 + DamageSource#getWeaponItem ⇒ 同一条派发链
            ItemStack bow = stack("chaosalloy_bow");
            int rangedHits = 0;
            int rn = 200;
            for (int i = 0; i < rn; i++) {
                target.removeAllEffects();
                target.setHealth(target.getMaxHealth());
                target.invulnerableTime = 0;
                Arrow arrow = new Arrow(level, player, new ItemStack(Items.ARROW), bow);
                DamageSource src = level.damageSources().arrow(arrow, player);
                target.hurt(src, 1.0F);
                if (target.hasEffect(AllEffects.POWER_IMBALANCE)
                        || target.hasEffect(AllEffects.RESISTANCE_IMBALANCE)) {
                    rangedHits++;
                }
            }
            double rratio = rangedHits / (double) rn;
            check("r7_imbalance_ranged_50pct", rratio > 0.40 && rratio < 0.60,
                    "n=" + rn + " hits=" + rangedHits + " ratio=" + num(rratio)
                            + " weaponItem=" + src_weapon_ok());

            // 盾牌：走同一条派发链（格挡反 buff）⇒ 器具侧那条抽样的同一入口 + 盾牌两档开关
            ItemStack shield = stack("chaosalloy_shield");
            boolean special = shield.getItem() instanceof MetalWeapons.MetalShieldItem ms && ms.isSpecialMetal();
            boolean immune = shield.getItem() instanceof MetalWeapons.MetalShieldItem ms2 && ms2.isBreakImmune();
            // ⚠ 免疫破盾的判据是「主 / 副手拿着这块盾」⇒ 必须先放进手里（首跑忘记放 ⇒ 假红）
            player.setItemSlot(EquipmentSlot.MAINHAND, shield);
            player.getCooldowns().addCooldown(shield.getItem(), 100);
            boolean beforeTick = player.getCooldowns().isOnCooldown(shield.getItem());
            NeoForge.EVENT_BUS.post(new PlayerTickEvent.Post(player));
            boolean cooled = player.getCooldowns().isOnCooldown(shield.getItem());
            player.setItemSlot(EquipmentSlot.MAINHAND, ItemStack.EMPTY);
            check("r7_shield_break_immune", special && immune && beforeTick && !cooled,
                    "specialMetal=" + special + " breakImmune=" + immune
                            + " cooldownBeforeTick=" + beforeTick + " cooldownAfterTick=" + cooled);
            check("r7_family_flags", fam.imbalanceOnAttack && fam.imbalanceReflect,
                    "imbalanceOnAttack=" + fam.imbalanceOnAttack + " imbalanceReflect=" + fam.imbalanceReflect);
        } catch (Throwable t) {
            fail("r7_imbalance", t.toString());
        }
    }

    // ---------- ⑤ 区间乒乓 + 攻速 + 挖掘 ----------
    private static void rangesAndSpeeds() {
        try {
            String[] forms = { "sword", "mace", "trident", "axe", "pickaxe", "shovel", "hoe", "knife" };
            List<String> bad = new ArrayList<>();
            for (String f : forms) {
                ItemStack st = stack("chaosalloy_" + f);
                ChaosRange.setHigh(st, ChaosRange.Channel.ATTACK, false);
                float low = (float) ChaosWiring.attackDamageOf(st);
                ChaosRange.setHigh(st, ChaosRange.Channel.ATTACK, true);
                float high = (float) ChaosWiring.attackDamageOf(st);
                MetalFamily fam = MetalFamily.of(st);
                var r = fam == null ? null : fam.range(st.getItem(), ChaosRange.Channel.ATTACK);
                // ⚠ 刀（乐事联动件）的属性由 FD 的 createKnife 建：**攻击加成 = tier 的 4.5 + knifeDamage**
                //   ⇒ 实机读数比声明值整体高 4.5（低端 10.0 / 高端 15.5，而声明是 5.5..11）。
                //   这不是轮7 引入的：混沌金轮3 的刀同一形状（声明 1.5..5.5 ⇒ 实机 6.0..10.0）。
                //   本轮**不改**（改了会同时改变已交付的混沌金刀），只在报告与规格里记账；
                //   判据只钉"区间**跨度**与声明一致"，并把那 4.5 的偏移显式写进读数。
                float offset = "knife".equals(f) ? 4.5F : 0.0F;
                if (r == null || Math.abs(low - (r.min() + offset)) > 1e-4
                        || Math.abs(high - (r.max() + offset)) > 1e-4) {
                    bad.add(f + "[" + num(low) + "/" + num(high) + " vs "
                            + (r == null ? "null" : num(r.min()) + "/" + num(r.max())) + "]");
                }
            }
            check("r7_ranges_pingpong_all", bad.isEmpty(),
                    "8 件近战装配的 低端/高端 读数与声明逐项相等（刀含 FD 的 +4.5 攻击加成偏移）bad=" + bad);

            // 弓 / 弩：弹射物读的是同一个 ATTACK 区间
            ItemStack bow = stack("chaosalloy_bow");
            ChaosRange.setHigh(bow, ChaosRange.Channel.ATTACK, false);
            float bowLow = MetalWeapons.MetalBowItem.bowBaseDamage(bow);
            ChaosRange.setHigh(bow, ChaosRange.Channel.ATTACK, true);
            float bowHigh = MetalWeapons.MetalBowItem.bowBaseDamage(bow);
            check("r7_bow_range_3_5_4_5", Math.abs(bowLow - 3.5F) < 1e-4 && Math.abs(bowHigh - 4.5F) < 1e-4,
                    "low=" + num(bowLow) + " high=" + num(bowHigh));
            check("r7_bow_full_draw_20", MetalWeapons.MetalBowItem.fullDrawTicks(bow) == 20,
                    "fullDrawTicks=" + MetalWeapons.MetalBowItem.fullDrawTicks(bow));
            ItemStack cb = stack("chaosalloy_crossbow");
            if (cb.getItem() instanceof MetalWeapons.MetalCrossbowItem) {
                int use = ((MetalWeapons.MetalCrossbowItem) cb.getItem()).getUseDuration(cb, player);
                check("r7_crossbow_charge_1_2", use == 27,
                        "getUseDuration=" + use + "（= chargeDuration 24 + 3；1.2 秒）");
            }

            // 挖掘速度：BreakSpeed 事件里补「当前值 − 上端」
            ItemStack pick = stack("chaosalloy_pickaxe");
            ChaosRange.setHigh(pick, ChaosRange.Channel.MINING, false);
            player.setItemSlot(EquipmentSlot.MAINHAND, pick);
            PlayerEvent.BreakSpeed ev = new PlayerEvent.BreakSpeed(player,
                    Blocks.STONE.defaultBlockState(), 16.0F, new BlockPos(baseX, baseY, baseZ));
            NeoForge.EVENT_BUS.post(ev);
            float lowSpeed = ev.getNewSpeed();
            ChaosRange.setHigh(pick, ChaosRange.Channel.MINING, true);
            PlayerEvent.BreakSpeed ev2 = new PlayerEvent.BreakSpeed(player,
                    Blocks.STONE.defaultBlockState(), 16.0F, new BlockPos(baseX, baseY, baseZ));
            NeoForge.EVENT_BUS.post(ev2);
            float highSpeed = ev2.getNewSpeed();
            check("r7_mining_range_8_16",
                    Math.abs(lowSpeed - 8.0F) < 1e-3 && Math.abs(highSpeed - 16.0F) < 1e-3,
                    "breakSpeed low=" + num(lowSpeed) + " high=" + num(highSpeed)
                            + "（original=16 = Tier 基准）");
        } catch (Throwable t) {
            fail("r7_ranges", t.toString());
        }
    }

    // ---------- ⑥ 重锤 ×1.4 / 盾牌 ----------
    private static void maceAndShield() {
        try {
            Zombie target = zombie(-6.0D, 0.0D);
            // 重锤：最终伤 ×1.4（同一层 LivingDamageEvent.Pre；显式 10 点便于复算）
            player.setItemSlot(EquipmentSlot.MAINHAND, stack("chaosalloy_mace"));
            target.removeAllEffects();
            target.setHealth(target.getMaxHealth());
            target.invulnerableTime = 0;
            float b = target.getHealth();
            target.hurt(level.damageSources().mobAttack(player), 10.0F);
            float maceLost = lost(b, target.getHealth());
            player.setItemSlot(EquipmentSlot.MAINHAND, stack("chaosalloy_sword"));
            // ⚠ 必须先清掉目标身上的失衡：崩乱合金器具命中会 50% 挂「抗性失衡」⇒
            //   第二发会被 ×2（轮7 实测：不清就是 20.0，而不是对照值 10.0）
            target.removeAllEffects();
            target.setHealth(target.getMaxHealth());
            target.invulnerableTime = 0;
            b = target.getHealth();
            target.hurt(level.damageSources().mobAttack(player), 10.0F);
            float swordLost = lost(b, target.getHealth());
            check("r7_mace_final_x1_4",
                    Math.abs(maceLost - 14.0F) < 1e-3 && Math.abs(swordLost - 10.0F) < 1e-3,
                    "mace=" + num(maceLost) + "（10 × 1.4） siblingSword=" + num(swordLost) + "（10，对照）");
        } catch (Throwable t) {
            fail("r7_mace", t.toString());
        }
    }

    // ---------- ⑦ 盔甲：4..10 区间减伤 + 每件 10% 反制 ----------
    private static void armor() {
        try {
            player.setItemSlot(EquipmentSlot.HEAD, stack("chaosalloy_helmet"));
            player.setItemSlot(EquipmentSlot.CHEST, stack("chaosalloy_chestplate"));
            player.setItemSlot(EquipmentSlot.LEGS, stack("chaosalloy_leggings"));
            player.setItemSlot(EquipmentSlot.FEET, stack("chaosalloy_boots"));
            List<Float> observed = new ArrayList<>();
            boolean formulaOk = true;
            for (int i = 0; i < 6; i++) {
                // 先读出四件的当前值之和，再打一发 magic（无视护甲）10 点
                float pct = 0.0F;
                for (EquipmentSlot s : new EquipmentSlot[] { EquipmentSlot.HEAD, EquipmentSlot.CHEST,
                        EquipmentSlot.LEGS, EquipmentSlot.FEET }) {
                    Float v = ChaosRange.current(player.getItemBySlot(s), MetalFamily.of(stack("chaosalloy_helmet")),
                            armorForm(s), ChaosRange.Channel.ARMOR);
                    if (v != null) {
                        pct += v / 100.0F;
                    }
                }
                player.setHealth(player.getMaxHealth());
                player.invulnerableTime = 0;
                player.hurtTime = 0;
                float b = player.getHealth();
                // ⚠ magic 会被 ServerPlayer 的 spawnInvulnerableTime 拦掉（首跑实测 hurtApplied=false）
                //   ⇒ 用 genericKill（bypasses_invulnerability + bypasses_armor）：伤害一定落地，
                //     而"区间减伤"那一步是 LivingDamageEvent.Pre 里的独立乘子，与伤害标签无关。
                boolean applied = player.hurt(level.damageSources().genericKill(), 10.0F);
                float got = lost(b, player.getHealth());
                float want = 10.0F * (1.0F - pct);
                observed.add(got);
                if (!applied || Math.abs(got - want) > 1e-3) {
                    formulaOk = false;
                }
                if (i == 0) {
                    bettergold.LOGGER.info("{} INFO armorDiag :: hurtApplied={} invulnerable={} abilitiesInv={} "
                                    + "isInvulnerableTo={} health={} pct={}", TAG, applied,
                            player.isInvulnerable(), player.getAbilities().invulnerable,
                            player.isInvulnerableTo(level.damageSources().magic()),
                            player.getHealth(), num(pct));
                }
            }
            check("r7_armor_range_reduction", formulaOk
                            && observed.stream().distinct().count() >= 2,
                    "6 发读数=" + observed + "（公式 = 10 × (1 − 四件当前值之和)；出现 >= 2 个不同值 = 在乒乓）");

            // 每件 10% ⇒ 四件 40%：大样本
            Zombie attacker = zombie(-9.0D, 0.0D);
            int n = 400;
            int procs = 0;
            for (int i = 0; i < n; i++) {
                attacker.removeAllEffects();
                player.setHealth(player.getMaxHealth());
                player.invulnerableTime = 0;
                player.hurtTime = 0;
                player.hurt(level.damageSources().mobAttack(attacker), 1.0F);
                if (attacker.hasEffect(AllEffects.POWER_IMBALANCE)
                        || attacker.hasEffect(AllEffects.RESISTANCE_IMBALANCE)) {
                    procs++;
                }
            }
            double ratio = procs / (double) n;
            check("r7_armor_reflect_40pct", ratio > 0.32 && ratio < 0.48,
                    "n=" + n + " procs=" + procs + " ratio=" + num(ratio) + "（4 × 10% = 40%）");
        } catch (Throwable t) {
            fail("r7_armor", t.toString());
        }
    }

    private static int readIntField(Object target, String field) {
        try {
            var f = target.getClass().getField(field);
            return f.getInt(target);
        } catch (Throwable t) {
            return -999;
        }
    }

    private static void clearSpawnInvulnerable(ServerPlayer p) {
        for (Class<?> c = p.getClass(); c != null && c != Object.class; c = c.getSuperclass()) {
            try {
                var f = c.getDeclaredField("spawnInvulnerableTime");
                f.setAccessible(true);
                f.setInt(p, 0);
                return;
            } catch (Throwable ignored) {
                // 继续往父类找
            }
        }
    }

    private static MetalFamily.Form armorForm(EquipmentSlot slot) {
        return switch (slot) {
            case HEAD -> MetalFamily.Form.HELMET;
            case CHEST -> MetalFamily.Form.CHESTPLATE;
            case LEGS -> MetalFamily.Form.LEGGINGS;
            default -> MetalFamily.Form.BOOTS;
        };
    }

    // ---------- ⑧ 崩乱禁果：饱和 11.6 / 16 分钟 / 1..4 级 / 冷却 ----------
    private static void fruit() {
        try {
            ItemStack fruit = stack("chaosalloy_forbidden_fruit");
            FoodProperties food = fruit.get(net.minecraft.core.component.DataComponents.FOOD);
            check("r7_fruit_nutrition_saturation",
                    food != null && food.nutrition() == 4 && Math.abs(food.saturation() - 11.6F) < 1e-4,
                    "nutrition=" + (food == null ? "-" : food.nutrition())
                            + " saturation=" + (food == null ? "-" : num(food.saturation()))
                            + "（原版金苹果 9.6 + 2 = 11.6）");

            // ⚠ 必须**先把 Item 抓出来**：吃完那一下 `ItemStack` 数量归零 ⇒ `getItem()` 变成 AIR，
            //   拿 `fruit.getItem()` 去查冷却会永远 false（轮7 实测踩过，A 级假红一次）。
            var fruitItem = fruit.getItem();
            player.removeAllEffects();
            player.getFoodData().setFoodLevel(10);
            player.getFoodData().setSaturation(0.0F);
            player.getCooldowns().removeCooldown(fruitItem);
            float satBefore = player.getFoodData().getSaturationLevel();
            fruit.finishUsingItem(level, player);
            float satAfter = player.getFoodData().getSaturationLevel();
            boolean cooled = player.getCooldowns().isOnCooldown(fruitItem);
            // 诊断：手动给一个无关物品加冷却，看这套 ItemCooldowns 在这个玩家身上是否工作
            player.getCooldowns().addCooldown(Items.APPLE, 100);
            boolean controlCd = player.getCooldowns().isOnCooldown(Items.APPLE);
            player.getCooldowns().removeCooldown(Items.APPLE);
            bettergold.LOGGER.info("{} INFO fruitDiag :: fruitCd={} controlCd={}", TAG, cooled, controlCd);
            check("r7_fruit_eat_saturation", Math.abs((satAfter - satBefore) - 11.6F) < 1e-3 && cooled,
                    "saturationDelta=" + num(satAfter - satBefore) + " cooldown=" + cooled);

            boolean found = false;
            String story = "-";
            for (var inst : player.getActiveEffects()) {
                story += inst.getEffect().value().getDisplayName().getString()
                        + "[amp=" + inst.getAmplifier() + ",dur=" + inst.getDuration() + "] ";
                if (inst.getDuration() == ChaosItems.CHAOSALLOY_FRUIT_EFFECT_TICKS
                        && inst.getAmplifier() >= 0 && inst.getAmplifier() <= 3) {
                    found = true;
                }
            }
            check("r7_fruit_effect_16min_1to4", found, "effects=" + story);
        } catch (Throwable t) {
            fail("r7_fruit", t.toString());
        }
    }
