#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Temporary perturbation harness (bg-15w §8.1/§8.3 static checkers).

For each case: perturb one source file on disk, run the validator, require exit 1 AND
the expected substring in stdout, then restore the original bytes and verify the SHA256
is back to the original. Baseline (no perturbation) must exit 0.

Pure ASCII on purpose (PowerShell / console encodings). Temporary: deleted at wrap-up.
"""
import hashlib
import os
import subprocess
import sys
from pathlib import Path

REPO = Path(__file__).resolve().parents[1]
JAVA = REPO / "src" / "main" / "java" / "com" / "hjmmd_8" / "bettergold"
CREATIVE = JAVA / "material" / "CreativeSections.java"
METAL_EVENTS = JAVA / "material" / "MetalEvents.java"
METAL_FAMILY = JAVA / "material" / "MetalFamily.java"
VALIDATOR = REPO / "tools" / "asset-generator" / "validate_metal_data.py"

ENV = dict(os.environ)
ENV["PYTHONIOENCODING"] = "utf-8"


def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def run_validator() -> tuple:
    p = subprocess.run([sys.executable, str(VALIDATOR)], capture_output=True, cwd=str(REPO), env=ENV)
    out = (p.stdout or b"").decode("utf-8", "replace") + (p.stderr or b"").decode("utf-8", "replace")
    return p.returncode, out


def sub_once(text: str, old: str, new: str) -> str:
    if text.count(old) != 1:
        raise SystemExit("PERTURB TARGET NOT UNIQUE: %r count=%d" % (old, text.count(old)))
    return text.replace(old, new)


CASES = [
    # name, file, transform(text) -> text, expected substring in validator stdout
    ("A GEAR_SLOT mace/axe ranks swapped back",
     CREATIVE,
     lambda t: sub_once(sub_once(t, 'Map.entry("mace", 1),', 'Map.entry("mace", 10),'),
                        'Map.entry("axe", 10),', 'Map.entry("axe", 1),'),
     "GEAR_SLOT"),
    ("B GEAR_SLOT hoe dropped (renamed to shears)",
     CREATIVE,
     lambda t: sub_once(t, 'Map.entry("hoe", 13)', 'Map.entry("shears", 13)'),
     "GEAR_SLOT"),
    ("C swim id literal duplicated (two ids = double)",
     METAL_EVENTS,
     lambda t: sub_once(t, '"swim_speed_water");',
                        '"swim_speed_water");\n    private static final String DUMMY_DUP_ID = "swim_speed_water";'),
     "2"),
    ("D ADD_MULTIPLIED_TOTAL reverted to ADD_VALUE",
     METAL_EVENTS,
     lambda t: sub_once(t, 'Operation.ADD_MULTIPLIED_TOTAL));', 'Operation.ADD_VALUE));'),
     "ADD_MULTIPLIED_TOTAL"),
    ("E MOVEMENT_SPEED added into getDefaultAttributeModifiers",
     METAL_FAMILY,
     lambda t: sub_once(t, 'net.minecraft.world.entity.ai.attributes.Attributes.WATER_MOVEMENT_EFFICIENCY,',
                        'net.minecraft.world.entity.ai.attributes.Attributes.MOVEMENT_SPEED,\n                    net.minecraft.world.entity.ai.attributes.Attributes.WATER_MOVEMENT_EFFICIENCY,'),
     "getDefaultAttributeModifiers"),
    ("F SWIM_SPEED_WATER_PER_PIECE 0.25F -> 0.5F",
     METAL_EVENTS,
     lambda t: sub_once(t, 'SWIM_SPEED_WATER_PER_PIECE = 0.25F', 'SWIM_SPEED_WATER_PER_PIECE = 0.5F'),
     "0.25F"),
]

lines = []


def emit(s: str) -> None:
    print(s)
    lines.append(s)


code, out = run_validator()
emit("OK    baseline validate_metal_data                 exit=%d" % code)
if code != 0:
    emit("ABORT baseline is not green; tail: " + out[-800:])
    Path(REPO / "build" / "bg8-perturb.txt").write_text("\n".join(lines) + "\n", encoding="utf-8")
    raise SystemExit(1)

failures = 0
for name, path, transform, expect in CASES:
    original = path.read_bytes()
    before = sha(path)
    text = original.decode("utf-8")
    new_text = transform(text)
    if new_text == text:
        emit("BAD   %s :: perturbation had no effect" % name)
        failures += 1
        continue
    path.write_bytes(new_text.encode("utf-8"))
    try:
        code, out = run_validator()
    finally:
        path.write_bytes(original)
    restored = sha(path)
    if restored != before:
        emit("BAD   %s :: restore FAILED (sha mismatch)" % name)
        failures += 1
        continue
    hit = expect in out
    ok = (code == 1 and hit)
    emit("%s %-52s exit=%d expect_hit=%s (%s)"
         % ("OK   " if ok else "BAD  ", name, code, hit, expect))
    if not ok:
        failures += 1
        emit("      tail: " + out[-600:].replace("\n", " | "))

code, out = run_validator()
emit("OK    baseline after restore validate_metal_data   exit=%d" % code)
if code != 0:
    failures += 1

Path(REPO / "build" / "bg8-perturb.txt").write_text("\n".join(lines) + "\n", encoding="utf-8")
emit("mismatches = %d" % failures)
raise SystemExit(1 if failures else 0)
