package com.hjmmd_8.bettergold.probe.bg22;

import java.lang.reflect.Field;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;
import java.util.UUID;

import com.hjmmd_8.bettergold.material.AllMetals;
import com.hjmmd_8.bettergold.material.MetalFamily;
import com.mojang.authlib.GameProfile;

import net.minecraft.core.BlockPos;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.level.ClientInformation;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.entity.EquipmentSlot;
import net.minecraft.world.entity.LivingEntity;
import net.minecraft.world.item.Item;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.level.block.Blocks;
import net.minecraft.world.level.block.SweetBerryBushBlock;
import net.neoforged.bus.api.SubscribeEvent;
import net.neoforged.fml.common.EventBusSubscriber;
import net.neoforged.neoforge.event.tick.ServerTickEvent;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

/**
 * bg22（1.7 收尾轮 A 级）临时服务端探针 —— <b>收尾整块删除</b>。
 *
 * <p>作者原话：「让树棘金盔甲不会因为甜浆果伤害掉耐久」。六个用例，全部用<b>真
 * {@link ServerPlayer}</b> + <b>真 {@code minecraft:sweet_berry_bush} 方块</b>（AGE=3）的
 * {@code BlockState#entityInside}（= {@code Entity#checkInsideBlocks} 里那一句）：</p>
 * <ol>
 *   <li>{@code berry_thorns4}：树棘金 4 件 + 甜浆果丛 ⇒ <b>4 件耐久都不变</b>、生命也不掉（100% 减免）。</li>
 *   <li>{@code berry_thorns1}：树棘金 1 件 + 甜浆果丛 ⇒ 该件耐久不变、生命照掉（减伤仍 25%）。</li>
 *   <li>{@code berry_echo4}：<b>别的族</b>（幽咆金）4 件 + 甜浆果丛 ⇒ <b>耐久各 +1</b>（负向）。</li>
 *   <li>{@code berry_mixed}：2 树棘金 + 2 幽咆金 + 甜浆果丛 ⇒ 只有树棘金两件不变、另外两件 +1
 *       （逐件粒度，不是"整次取消"）。</li>
 *   <li>{@code mobattack_thorns4}：树棘金 4 件 + <b>别的伤害源</b>（{@code minecraft:mob_attack}，真僵尸当攻击者）⇒
 *       <b>耐久各 +1</b>（负向；⚠ 不用 cactus：树棘金本来就免疫仙人掌 §3.8，那是另一条既有口径；
 *       ⚠ 也不用 magic：1.21.1 的 {@code #minecraft:bypasses_armor} 里就有 {@code magic}
 *       ⇒ 它<b>根本不走 hurtArmor</b>，会把这条负向测成假绿）。</li>
 *   <li>{@code cactus_thorns4}：树棘金 4 件 + 仙人掌 ⇒ 耐久不变（= 既有 handler 没被本轮改坏）。</li>
 * </ol>
 *
 * <p>开关文件 {@code run/bg22-probe.enabled}；forceload 双向归零；{@code halt(false)} 只调一次。
 * 读数落 {@code run/probe/bg22/readings.txt}（收尾归档到 {@code docs/bg22-证据/}）。</p>
 */
@EventBusSubscriber(modid = "bettergold")
public final class Bg22Probe {

    private static final String TAG = "[bg22-s]";
    private static final Path SWITCH = Path.of("bg22-probe.enabled");
    private static final Path OUT = Path.of("probe", "bg22", "readings.txt");
    private static final Logger LOG = LoggerFactory.getLogger("Bg22Probe");

    private static final BlockPos BASE = new BlockPos(2000, 120, 2000);
    private static final BlockPos BUSH = BASE;
    private static final int FORCED_X = BASE.getX() >> 4;
    private static final int FORCED_Z = BASE.getZ() >> 4;

    private static final List<String> LINES = new ArrayList<>();
    private static int passes;
    private static int fails;
    private static int phase;
    private static int tick;
    private static boolean done;
    private static ServerLevel level;
    private static ServerPlayer man;

    private Bg22Probe() {
    }

    private static void check(String id, boolean ok, String detail) {
        if (ok) {
            passes++;
            LINES.add("PASS " + id + " :: " + detail);
            LOG.info("{} PASS {} :: {}", TAG, id, detail);
        } else {
            fails++;
            LINES.add("FAIL " + id + " :: " + detail);
            LOG.error("{} FAIL {} :: {}", TAG, id, detail);
        }
    }

    @SubscribeEvent
    public static void onTick(ServerTickEvent.Post event) {
        if (done) {
            return;
        }
        MinecraftServer server = event.getServer();
        if (phase == 0) {
            if (!Files.isRegularFile(SWITCH)) {
                return;
            }
            try {
                setup(server);
                phase = 1;
                tick = 0;
            } catch (Throwable t) {
                LOG.error("{} setup threw", TAG, t);
                check("probe_setup", false, String.valueOf(t));
                finish(server);
            }
            return;
        }
        if (phase == 1) {
            if (++tick < 10) {
                return;   // 等 chunk 到 FULL / 实体管理器就绪（bg18b 踩过的假阴性）
            }
            try {
                runCases();
            } catch (Throwable t) {
                LOG.error("{} measure threw", TAG, t);
                check("probe_measure", false, String.valueOf(t));
            }
            phase = 2;
            tick = 0;
            return;
        }
        if (phase == 2) {
            if (++tick < 3) {
                return;
            }
            finish(server);
        }
    }

    // ==================== setup ====================

    private static void setup(MinecraftServer server) {
        level = server.overworld();
        level.setChunkForced(FORCED_X, FORCED_Z, true);
        level.getChunk(FORCED_X, FORCED_Z);
        for (int dx = -2; dx <= 2; dx++) {
            for (int dz = -2; dz <= 2; dz++) {
                level.setBlockAndUpdate(new BlockPos(BASE.getX() + dx, BASE.getY() - 1, BASE.getZ() + dz),
                        Blocks.STONE.defaultBlockState());
                level.setBlockAndUpdate(new BlockPos(BASE.getX() + dx, BASE.getY(), BASE.getZ() + dz),
                        Blocks.AIR.defaultBlockState());
                level.setBlockAndUpdate(new BlockPos(BASE.getX() + dx, BASE.getY() + 1, BASE.getZ() + dz),
                        Blocks.AIR.defaultBlockState());
            }
        }
        level.setBlockAndUpdate(BUSH, Blocks.SWEET_BERRY_BUSH.defaultBlockState()
                .setValue(SweetBerryBushBlock.AGE, Integer.valueOf(3)));

        man = player(server, "bg22_armor");
        man.moveTo(BUSH.getX() + 0.5D, BUSH.getY(), BUSH.getZ() + 0.5D, 0.0F, 0.0F);
        man.setHealth(20.0F);
        level.addNewPlayer(man);

        LINES.add("SETUP level=" + levelName(server) + " bush=" + BUSH.toShortString()
                + " forced=" + level.getForcedChunks().size() + " player=" + man.getGameProfile().getName());
        LOG.info("{} readout setup level={} bush={} forced={}", TAG, levelName(server),
                BUSH.toShortString(), level.getForcedChunks().size());
    }

    private static String levelName(MinecraftServer server) {
        try {
            Object data = server.overworld().getLevelData();
            var m = data.getClass().getMethod("getLevelName");
            return String.valueOf(m.invoke(data));
        } catch (Throwable t) {
            return "<unreadable:" + t.getClass().getSimpleName() + ">";
        }
    }

    // ==================== 六个用例 ====================

    private static void runCases() {
        var bush = level.getBlockState(BUSH);

        // ① 树棘金 4 件 + 甜浆果丛：耐久全不变、生命也不掉
        wear(AllMetals.THORNSGOLD, 4);
        resetHurt();
        int[] before = damages();
        bush.entityInside(level, BUSH, man);
        int[] a = damages();
        check("berry_thorns4_durability_unchanged", same(a, before),
                "before=" + arr(before) + " after=" + arr(a) + " hp=" + man.getHealth());
        check("berry_thorns4_health_unchanged", Math.abs(man.getHealth() - 20.0F) < 1.0E-4F,
                "hp=" + man.getHealth() + " (want 20.000)");

        // ② 树棘金 1 件 + 甜浆果丛：那一件耐久不变，生命照掉
        wear(AllMetals.THORNSGOLD, 1);
        resetHurt();
        before = damages();
        bush.entityInside(level, BUSH, man);
        a = damages();
        check("berry_thorns1_durability_unchanged", same(a, before),
                "before=" + arr(before) + " after=" + arr(a) + " hp=" + man.getHealth());
        check("berry_thorns1_health_dropped", man.getHealth() < 20.0F,
                "hp=" + man.getHealth() + " (want < 20)");

        // ③ 别的族 4 件 + 甜浆果丛：耐久各 +1（负向）
        wear(AllMetals.ECHOGOLD, 4);
        resetHurt();
        before = damages();
        bush.entityInside(level, BUSH, man);
        a = damages();
        check("berry_echo4_durability_hurt", deltaIs(a, before, 1),
                "before=" + arr(before) + " after=" + arr(a) + " hp=" + man.getHealth());
        check("berry_echo4_health_dropped", man.getHealth() < 20.0F,
                "hp=" + man.getHealth() + " (want < 20)");

        // ④ 混合套装：2 树棘金 + 2 幽咆金 ⇒ 只有树棘金两件不变
        wearMixed(2);
        resetHurt();
        before = damages();
        bush.entityInside(level, BUSH, man);
        a = damages();
        check("berry_mixed_thorns_unchanged",
                a[0] == before[0] && a[1] == before[1],
                "head/chest(树棘金) " + before[0] + "," + before[1] + " -> " + a[0] + "," + a[1]);
        check("berry_mixed_other_family_hurt",
                a[2] == before[2] + 1 && a[3] == before[3] + 1,
                "legs/feet(幽咆金) " + before[2] + "," + before[3] + " -> " + a[2] + "," + a[3] + " (want +1)");
        check("berry_mixed_health_dropped", man.getHealth() < 20.0F,
                "hp=" + man.getHealth() + " (want < 20)");

        // ⑤ 树棘金 4 件 + 别的伤害源（真僵尸的 mob_attack）：耐久各 +1（负向）
        //   ⚠ 为什么不是 magic：`#minecraft:bypasses_armor` 里就有 magic（client jar 实测）
        //     ⇒ 它连 hurtArmor 都不走，测不出「别的伤害源照扣」。
        var zombie = net.minecraft.world.entity.EntityType.ZOMBIE.create(level);
        if (zombie == null) {
            throw new IllegalStateException("cannot create zombie");
        }
        zombie.moveTo(BUSH.getX() + 2.5D, BUSH.getY(), BUSH.getZ() + 0.5D, 0.0F, 0.0F);
        level.addFreshEntity(zombie);
        wear(AllMetals.THORNSGOLD, 4);
        resetHurt();
        before = damages();
        man.hurt(level.damageSources().mobAttack(zombie), 1.0F);
        a = damages();
        check("mobattack_thorns4_durability_hurt", deltaIs(a, before, 1),
                "before=" + arr(before) + " after=" + arr(a) + " hp=" + man.getHealth());
        check("mobattack_thorns4_health_dropped", man.getHealth() < 20.0F,
                "hp=" + man.getHealth() + " (want < 20)");
        zombie.discard();

        // ⑥ 树棘金 4 件 + 仙人掌：既有 handler 原样（耐久仍不变）
        wear(AllMetals.THORNSGOLD, 4);
        resetHurt();
        before = damages();
        man.hurt(level.damageSources().cactus(), 1.0F);
        a = damages();
        check("cactus_thorns4_durability_unchanged", same(a, before),
                "before=" + arr(before) + " after=" + arr(a) + " hp=" + man.getHealth());
        check("cactus_thorns4_health_unchanged", Math.abs(man.getHealth() - 20.0F) < 1.0E-4F,
                "hp=" + man.getHealth() + " (want 20.000)");

        wear(AllMetals.THORNSGOLD, 0);
    }

    // ==================== 小工具 ====================

    private static final EquipmentSlot[] SLOTS = {
            EquipmentSlot.HEAD, EquipmentSlot.CHEST, EquipmentSlot.LEGS, EquipmentSlot.FEET};

    /** 给玩家穿 n 件该族盔甲（头/胸/腿/靴 依次），其余槽位清空。 */
    private static void wear(MetalFamily family, int n) {
        Item[] pieces = {family.helmet.get(), family.chestplate.get(), family.leggings.get(), family.boots.get()};
        for (int i = 0; i < 4; i++) {
            man.setItemSlot(SLOTS[i], i < n ? new ItemStack(pieces[i]) : ItemStack.EMPTY);
        }
    }

    /** 头/胸 = 树棘金，腿/靴 = 幽咆金。 */
    private static void wearMixed(int thornsPieces) {
        Item[] thorns = {AllMetals.THORNSGOLD.helmet.get(), AllMetals.THORNSGOLD.chestplate.get(),
                AllMetals.THORNSGOLD.leggings.get(), AllMetals.THORNSGOLD.boots.get()};
        Item[] echo = {AllMetals.ECHOGOLD.helmet.get(), AllMetals.ECHOGOLD.chestplate.get(),
                AllMetals.ECHOGOLD.leggings.get(), AllMetals.ECHOGOLD.boots.get()};
        for (int i = 0; i < 4; i++) {
            man.setItemSlot(SLOTS[i], new ItemStack(i < thornsPieces ? thorns[i] : echo[i]));
        }
    }

    private static int[] damages() {
        int[] d = new int[4];
        for (int i = 0; i < 4; i++) {
            d[i] = man.getItemBySlot(SLOTS[i]).getDamageValue();
        }
        return d;
    }

    private static boolean same(int[] a, int[] b) {
        for (int i = 0; i < 4; i++) {
            if (a[i] != b[i]) {
                return false;
            }
        }
        return true;
    }

    private static boolean deltaIs(int[] after, int[] before, int want) {
        for (int i = 0; i < 4; i++) {
            if (after[i] - before[i] != want) {
                return false;
            }
        }
        return true;
    }

    private static String arr(int[] a) {
        return "[" + a[0] + "," + a[1] + "," + a[2] + "," + a[3] + "]";
    }

    /** 清出生无敌 + 无敌帧（否则 hurt 恒 false ⇒ 假阴性）。 */
    private static void resetHurt() {
        man.setHealth(20.0F);
        man.setAbsorptionAmount(0.0F);
        man.invulnerableTime = 0;
        man.hurtTime = 0;
        man.deathTime = 0;
        man.xOld = man.getX() - 0.1D;      // entityInside 的移动判定门槛（d0 >= 0.003）
        man.zOld = man.getZ();
        try {
            Field f = ServerPlayer.class.getDeclaredField("spawnInvulnerableTime");
            f.setAccessible(true);
            f.setInt(man, 0);
        } catch (Throwable ignored) {
            // ignore
        }
        try {
            Field f = LivingEntity.class.getDeclaredField("invulnerableTime");
            f.setAccessible(true);
            f.setInt(man, 0);
        } catch (Throwable ignored) {
            // ignore
        }
    }

    private static ServerPlayer player(MinecraftServer server, String name) {
        GameProfile profile = new GameProfile(
                UUID.nameUUIDFromBytes(("bg22:" + name).getBytes(StandardCharsets.UTF_8)), name);
        ServerPlayer p = new ServerPlayer(server, level, profile, ClientInformation.createDefault());
        try {
            p.connection = new net.minecraft.server.network.ServerGamePacketListenerImpl(
                    server,
                    new net.minecraft.network.Connection(
                            net.minecraft.network.protocol.PacketFlow.SERVERBOUND),
                    p,
                    net.minecraft.server.network.CommonListenerCookie.createInitial(profile, false));
        } catch (Throwable t) {
            LOG.warn("{} dummy connection failed for {}: {}", TAG, name, String.valueOf(t));
        }
        return p;
    }

    private static void finish(MinecraftServer server) {
        if (done) {
            return;
        }
        done = true;
        try {
            level.setChunkForced(FORCED_X, FORCED_Z, false);
        } catch (Throwable t) {
            LOG.warn("{} unforce threw {}", TAG, String.valueOf(t));
        }
        int forced = level == null ? -1 : level.getForcedChunks().size();
        check("forceload_zero", forced == 0, "forcedNow=" + forced);
        LINES.add("SUMMARY pass=" + passes + " fail=" + fails + " forcedNow=" + forced);
        LOG.info("{} SUMMARY pass={} fail={} forcedNow={}", TAG, passes, fails, forced);
        try {
            Files.createDirectories(OUT.getParent());
            Files.writeString(OUT, String.join("\n", LINES) + "\n", StandardCharsets.UTF_8);
        } catch (Throwable t) {
            LOG.error("{} cannot write {}: {}", TAG, OUT, String.valueOf(t));
        }
        server.halt(false);
    }
}
