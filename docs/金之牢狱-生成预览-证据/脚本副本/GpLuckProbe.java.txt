package com.hjmmd_8.bettergold.probe;

/**
 * 【临时探针 · 收尾整块删除】金之牢狱战利品「幸运 0 vs 3」回归（runServer）。
 *
 * 只做一件事：把 `bettergold:chests/golden_prison` 各 roll 100 箱，luck=0 与 luck=3 各算平均件数，
 * 打印 `[gpl] luck0=... luck3=... empty0=...` 供离线判读；只调一次 server.halt(false)。
 *
 * 开关：run/probe/gpx/PROBE_ON.flag 存在才跑（与预览探针共用同一个开关目录，两者不会同时进 src）。
 */
import com.mojang.logging.LogUtils;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.List;
import net.minecraft.core.registries.Registries;
import net.minecraft.resources.ResourceKey;
import net.minecraft.resources.ResourceLocation;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.util.RandomSource;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.level.storage.loot.LootParams;
import net.minecraft.world.level.storage.loot.LootTable;
import net.minecraft.world.level.storage.loot.parameters.LootContextParamSets;
import net.minecraft.world.level.storage.loot.parameters.LootContextParams;
import net.minecraft.world.phys.Vec3;
import net.neoforged.bus.api.SubscribeEvent;
import net.neoforged.fml.common.EventBusSubscriber;
import net.neoforged.neoforge.event.server.ServerStartedEvent;
import org.slf4j.Logger;

@EventBusSubscriber(modid = "bettergold")
public final class GpLuckProbe {
    private static final Logger LOG = LogUtils.getLogger();
    private static final Path OUT = Path.of("probe", "gpx");
    private static final ResourceLocation LOOT =
            ResourceLocation.fromNamespaceAndPath("bettergold", "chests/golden_prison");
    private static final int ROLLS = 100;

    private GpLuckProbe() {}

    @SubscribeEvent
    public static void onServerStarted(ServerStartedEvent event) {
        Path flag = OUT.resolve("PROBE_ON.flag");
        if (!Files.exists(flag)) {
            LOG.info("[gpl] probe OFF (no probe/gpx/PROBE_ON.flag)");
            return;
        }
        MinecraftServer server = event.getServer();
        try {
            ServerLevel level = server.overworld();
            LootTable lt = server.reloadableRegistries().getLootTable(
                    ResourceKey.create(Registries.LOOT_TABLE, LOOT));
            if (lt == null) {
                LOG.error("[gpl] loot table missing: {}", LOOT);
            } else {
                RandomSource rnd = level.getRandom();
                double[] avg = new double[2];
                int[] empty = new int[2];
                int[] maxn = new int[2];
                for (int k = 0; k < 2; k++) {
                    float luck = k == 0 ? 0.0F : 3.0F;
                    LootParams params = new LootParams.Builder(level)
                            .withParameter(LootContextParams.ORIGIN, Vec3.atCenterOf(level.getSharedSpawnPos()))
                            .withLuck(luck)
                            .create(LootContextParamSets.CHEST);
                    long total = 0;
                    for (int i = 0; i < ROLLS; i++) {
                        List<ItemStack> items = lt.getRandomItems(params, rnd);
                        int n = 0;
                        for (ItemStack st : items) {
                            n += st.getCount();
                        }
                        if (n == 0) {
                            empty[k]++;
                        }
                        maxn[k] = Math.max(maxn[k], n);
                        total += n;
                    }
                    avg[k] = total / (double) ROLLS;
                }
                LOG.info("[gpl] luck0={} luck3={} empty0={} empty3={} max0={} max3={} rolls={}",
                        String.format("%.2f", avg[0]), String.format("%.2f", avg[1]),
                        empty[0], empty[1], maxn[0], maxn[1], ROLLS);
                LOG.info("[gpl] PASS luckRaises={} neverEmpty={}",
                        avg[1] > avg[0], empty[0] == 0);
            }
        } catch (Throwable t) {
            LOG.error("[gpl] EXCEPTION", t);
        }
        LOG.info("[gpl] DONE");
        server.halt(false);
    }
}
