    @SubscribeEvent
    public static void onServerTick(ServerTickEvent.Post event) {
        if (!armed) {
            Path flag = FMLPaths.GAMEDIR.get().resolve("bgfix9s-probe.enabled");
            if (!Files.isRegularFile(flag)) {
                return;
            }
            armed = true;
            server = event.getServer();
            level = server.overworld();
            readout("probe_armed", "gamedir=" + FMLPaths.GAMEDIR.get() + " flag=" + flag
                    + " repoGuess=" + FMLPaths.GAMEDIR.get().getParent());
        }
        if (aborted) {
            return;
        }
        tick++;
        try {
            if (tick == 1) {
                guardWorld();
            } else if (tick == 40) {
                runAll();
            } else if (tick == 60) {
                summary();
            }
        } catch (Throwable t) {
            problems.add("tick=" + tick + " 抛异常：" + t);
            bettergold.LOGGER.error("{} tick={} 抛异常", TAG, tick, t);
            for (StackTraceElement e : t.getStackTrace()) {
                bettergold.LOGGER.error("{}   at {}", TAG, e);
            }
        }
    }

    private static void readout(String name, String value) {
        bettergold.LOGGER.info("{} {} :: {}", TAG, name, value);

    private static void check(String name, boolean ok, String detail) {
        checks++;
        readout(name, (ok ? "PASS" : "FAIL") + " :: " + detail);
        if (!ok) {
            problems.add(name + " :: " + detail);
        }
    }

    private static void guardWorld() {
        String name = "?";
        try {
            Object data = level.getLevelData();
            Method m = data.getClass().getMethod("getLevelName");
            name = String.valueOf(m.invoke(data));
        } catch (Throwable t) {
            name = "<反射失败:" + t.getClass().getSimpleName() + ">";
        }
        boolean ok = PROBE_WORLD.equals(name) && !name.contains(AUTHOR_WORLD);
        readout("probe_world_guard", (ok ? "PASS" : "FAIL") + " :: levelName=" + name
                + " expect=" + PROBE_WORLD + " authorWorld=" + AUTHOR_WORLD);
        if (!ok) {
            bettergold.LOGGER.error("{} ABORT 世界名不对 ⇒ 停止一切动作（不 halt，避免把当前世界存盘）", TAG);
            aborted = true;
        }
        readout("advancements_loaded", "本模组两条目标成就在注册表里 = "
                + (adv(ADV_RAW) != null) + " / " + (adv(ADV_EGG) != null)
                + "（真 ServerAdvancementManager）");
    }

        } catch (Throwable t) {
            readout("dummy_connection_" + tag, "装不上：" + t);

        // ★★ 第一轮实测的**最大陷阱**：`ServerPlayer` 的 `containerListener` 是 **private**，

        } catch (Throwable t) {
            readout("init_inventory_menu_" + tag, "FAIL :: " + t);
        }
        return p;
    }

    private static AdvancementHolder adv(String id) {
        return server.getAdvancements().get(ResourceLocation.parse(id));
    }

    private static boolean done(ServerPlayer p, String id) {
        AdvancementHolder h = adv(id);
        return h != null && p.getAdvancements().getOrStartProgress(h).isDone();
    }

    /** 用**游戏自己的**可见性评估器（`AdvancementVisibilityEvaluator`）判"客户端看不看得见"。 */
    private static boolean visible(ServerPlayer p, String id) {
        AdvancementHolder h = adv(id);
        if (h == null) {
            return false;
        }
        AdvancementNode node = server.getAdvancements().tree().get(h);
        if (node == null) {
            return false;
        }
        final boolean[] out = {false};
        AdvancementVisibilityEvaluator.evaluateVisibility(
                node,
                n -> p.getAdvancements().getOrStartProgress(n.holder()).isDone(),
                (n, vis) -> {
                    if (n.holder().id().equals(h.id())) {
                        out[0] = vis;
                    }
                });
        return out[0];
    }

    private static ItemStack stackOf(String id) {
        var item = BuiltInRegistries.ITEM.get(ResourceLocation.parse(id));
        return item == Items.AIR ? ItemStack.EMPTY : new ItemStack(item);
    }

    /**
     * 「拿到东西」= 物品落进玩家背包 ⇒ **真派发点**（`ServerPlayer#doTick` 里那句
     * {@code containerMenu.broadcastChanges()} → {@code ServerPlayer}'s ContainerListener#slotChanged
     * → {@code CriteriaTriggers.INVENTORY_CHANGED.trigger}）。
     *
     * <p>先把 `containerMenu` 复位成 `inventoryMenu`（唯一挂了 containerListener 的那个菜单）。</p>
     */
    private static void give(ServerPlayer p, ItemStack st) {
        p.containerMenu = p.inventoryMenu;
        p.getInventory().add(st);
        p.containerMenu.broadcastChanges();
    }

    private static int[] invCounts(Inventory inv) {
        int full = 0;
        int empty = 0;
        int occupied = 0;
        for (int i = 0; i < inv.getContainerSize(); i++) {
            ItemStack s = inv.getItem(i);
            if (s.isEmpty()) {
                empty++;
            } else {
                occupied++;
                if (s.getCount() >= s.getMaxStackSize()) {
                    full++;
                }
            }
        }
        return new int[]{full, empty, occupied};
    }

    /** 用**真 codec** 把产物 JSON / 历史 JSON 解析成判据实例（不是手搓等价物）。
     *
     * <p>⚠ 第一轮实测陷阱：`ItemPredicate.CODEC` 里的 `RegistryCodecs.homogeneousList(Registries.ITEM)`
     * 需要 **`RegistryOps`**（纯 `JsonOps` 下 `.parse()` 直接返回 error、`.result()` 为空 ⇒ 静默 null）。
     * 所以这里必须 `RegistryOps.create(JsonOps.INSTANCE, server.registryAccess())`，并把 error 打进日志。</p>
     */
    private static InventoryChangeTrigger.TriggerInstance parseHave(String relPath, String critKey) {
        try {
            Path repo = FMLPaths.GAMEDIR.get().getParent();
            String txt = Files.readString(repo.resolve(relPath), StandardCharsets.UTF_8);
            JsonObject root = JsonParser.parseString(txt).getAsJsonObject();
            JsonObject cond = root.getAsJsonObject("criteria").getAsJsonObject(critKey)
                    .getAsJsonObject("conditions");
            RegistryOps<com.google.gson.JsonElement> ops =
                    RegistryOps.create(JsonOps.INSTANCE, server.registryAccess());
            var res = InventoryChangeTrigger.TriggerInstance.CODEC.parse(ops, cond);
            if (res.result().isEmpty()) {
                readout("parse_fail", relPath + " -> " + res.error().map(Object::toString).orElse("?"));
                return null;
            }
            return res.result().get();
        } catch (Throwable t) {
            readout("parse_fail", relPath + " -> " + t);
            return null;
        }
    }

    private static boolean triggerMatches(InventoryChangeTrigger.TriggerInstance inst, ServerPlayer p,
                                          ItemStack changed) {
        int[] c = invCounts(p.getInventory());
        return inst.matches(p.getInventory(), changed, c[0], c[1], c[2]);
    }

    /** 判据里那条 `player`（`ContextAwarePredicate`）—— 与 `SimpleCriterionTrigger#trigger` 同一处判定。 */
    private static boolean playerCondOk(InventoryChangeTrigger.TriggerInstance inst, ServerPlayer p) {
        Optional<?> opt = inst.player();
        if (opt.isEmpty()) {
            return true;
        }
        LootContext ctx = EntityPredicate.createContext(p, p);
        return inst.player().get().matches(ctx);
    }

    // ---- ① A：拿取 / 合成 ----
    private static void partA() {
        String rawId = "bettergold:raw_flamegold";

        ServerPlayer pa = newPlayer("a-pickup");
        give(pa, stackOf(rawId));
        check("A1_pickup_unlock", done(pa, ADV_RAW),
                "拿取 1 个 " + rawId + " ⇒ " + ADV_RAW + ".isDone=" + done(pa, ADV_RAW)
                        + "（旧形状要求同时持有 8 种）");

        ServerPlayer pc = newPlayer("a-craft");
        String craftInfo = craftRawMetal(pc);
        boolean landed = false;
        for (int i = 0; i < pc.getInventory().getContainerSize(); i++) {
            if (pc.getInventory().getItem(i).is(
                    BuiltInRegistries.ITEM.get(ResourceLocation.parse("bettergold:raw_flamegold")))) {
                landed = true;
            }
        }
        check("A2_craft_unlock", done(pc, ADV_RAW) && landed,
                "合成路径（真 CraftingMenu + 真两步点击；产物真的落进背包=" + landed + "）" + craftInfo
                        + " ⇒ isDone=" + done(pc, ADV_RAW));

        // 改前 / 改后：同一份真背包，两种判据形状各自求值
        ServerPlayer pj = newPlayer("a-judge");
        give(pj, stackOf(rawId));
        ItemStack changed = pj.getInventory().getItem(0);
        InventoryChangeTrigger.TriggerInstance oldInst = parseHave(
                "docs/bgfix9-\u8bc1\u636e/_adv-before/treasure/any_raw_metal.json", "have");
        InventoryChangeTrigger.TriggerInstance newInst = parseHave(
                "src/main/resources/data/bettergold/advancement/treasure/any_raw_metal.json", "have");
        boolean old1 = oldInst != null && triggerMatches(oldInst, pj, changed);
        boolean new1 = newInst != null && triggerMatches(newInst, pj, changed);
        check("A3_one_raw_old_false_new_true", !old1 && new1,
                "只持有 1 种原料：旧形状 matches=" + old1 + " / 现行形状 matches=" + new1
                        + "（旧 JSON = " + (oldInst == null ? "解析失败" : "解析成功") + "）");

        // 把 8 种原料都塞进背包 ⇒ 旧形状才该变 true（= 证明旧形状是 AND-of-8）
        for (String m : new String[]{"bettergold:raw_voodoogold", "bettergold:raw_thundergold",
                "bettergold:raw_indigoseagold", "bettergold:raw_illusiongold",
                "bettergold:raw_thornsgold", "bettergold:raw_echogold", "bettergold:raw_sturdygold"}) {
            pj.getInventory().add(stackOf(m));
        }
        boolean old8 = oldInst != null && triggerMatches(oldInst, pj, changed);
        boolean new8 = newInst != null && triggerMatches(newInst, pj, changed);
        check("A4_eight_raw_old_true", old8 && new8,
                "持有全部 8 种原料：旧形状 matches=" + old8 + " / 现行形状 matches=" + new8
                        + "（旧形状 = 「同时持有全部 8 种」⇒ 这就是作者「两条路径都触发不了」的原因）");
    }

    /**
     * 真合成路径：真 {@code CraftingMenu}（必须给**真** {@code ContainerLevelAccess}，
     * 否则 {@code slotsChanged} 里的 {@code access.execute(...)} 是空操作、产物永远为空）
     * + 两次**真点击**（结果格 → 光标 → 背包空槽）。
     *
     * <p>第一轮实测：结果格一直是 air（而同一份 3×3 输入用 `RecipeManager` 单独查是查得到的）
     * ⇒ 本轮加了三条诊断（格内读回 / `doLimitedCrafting` / 用**菜单自己的** `craftSlots`
     * 再查一次配方），并保留一条**兜底**：真配方 + 真 `assemble` + 真背包派发。</p>
     */
    private static String craftRawMetal(ServerPlayer p) {
        BlockPos pos = new BlockPos(400, 200, 380);
        level.setBlockAndUpdate(pos.below(), Blocks.STONE.defaultBlockState());
        // 诊断①：证明真 access 的 execute 真的会跑 lambda（NULL access 是空操作）
        ContainerLevelAccess.create(level, pos).execute((lv, ps) ->
                readout("access_execute_ran", "PASS :: ContainerLevelAccess.create(level,pos).execute 的 lambda 已执行"));
        CraftingMenu menu = new CraftingMenu(0, p.getInventory(), ContainerLevelAccess.create(level, pos));
        p.containerMenu = menu;
        ItemStack[] ing = {
                new ItemStack(AllItems.MIXED_CRYSTAL_PILE.get()), new ItemStack(AllItems.MIXED_CRYSTAL_PILE.get()),
                new ItemStack(Items.REDSTONE), new ItemStack(Items.REDSTONE),
                new ItemStack(Items.GOLD_BLOCK), new ItemStack(Items.GOLD_BLOCK),
                new ItemStack(Items.NETHERITE_INGOT), new ItemStack(AllItems.ALCHEMIC_FUEL.get()),
                stackOf("bettergold:blazing_rod"),
        };
        for (int i = 0; i < 9; i++) {
            menu.getSlot(1 + i).set(ing[i]);
        }
        menu.slotsChanged(menu.getSlot(1).container);
        ItemStack r1 = menu.getSlot(0).getItem().copy();
        StringBuilder grid = new StringBuilder();
        for (int i = 0; i < 9; i++) {
            grid.append(BuiltInRegistries.ITEM.getKey(menu.getSlot(1 + i).getItem().getItem()).getPath()).append(' ');
        }
        CraftingInput menuInput = ((net.minecraft.world.inventory.CraftingContainer)
                menu.getSlot(1).container).asCraftInput();
        String menuQuery = server.getRecipeManager().getRecipeFor(RecipeType.CRAFTING, menuInput, level)
                .map(h -> h.id().toString()).orElse("<无配方>");
        boolean limited = level.getGameRules()
                .getBoolean(net.minecraft.world.level.GameRules.RULE_LIMITED_CRAFTING);
        String managerResult;
        ItemStack assembled = ItemStack.EMPTY;
        try {
            CraftingInput input = CraftingInput.of(3, 3, List.of(ing));
            var holder = server.getRecipeManager().getRecipeFor(RecipeType.CRAFTING, input, level);
            managerResult = holder
                    .map(h -> h.id() + " -> " + h.value().getResultItem(level.registryAccess()).getItem())
                    .orElse("<无配方>");
            if (holder.isPresent()) {
                assembled = holder.get().value().assemble(input, level.registryAccess());
            }
        } catch (Throwable t) {
            managerResult = "<查询失败:" + t + ">";
        }
        // 诊断②：`slotsChanged` 没算出产物时，按 `slotChangedCraftingGrid` 的**同一条式子**补结果格
        //   （真 RecipeManager + 真 assemble；这样"两次真点击"这条 UI 路径仍然是真的）
        String refill = "<不需要>";
        if (r1.isEmpty() && !assembled.isEmpty()) {
            menu.getSlot(0).container.setItem(0, assembled.copy());
            refill = "已按 slotChangedCraftingGrid 的同一条式子补上（" + assembled.getCount() + "x）";
        }
        ItemStack result = menu.getSlot(0).getItem().copy();
        int free = -1;
        for (int i = 10; i <= 45; i++) {
            if (menu.getSlot(i).getItem().isEmpty()) {
                free = i;
                break;
            }
        }
        menu.clicked(0, 0, ClickType.PICKUP, p);          // 结果格 → 光标（内部会 onTake + 扣原料）
        menu.clicked(free, 0, ClickType.PICKUP, p);       // 光标 → 背包空槽（= 真玩家第二步点击）
        // 真派发：换回挂了 containerListener 的 inventoryMenu，再走一次真 broadcastChanges
        p.containerMenu = p.inventoryMenu;
        p.containerMenu.broadcastChanges();
        ItemStack landed = ItemStack.EMPTY;
        for (int i = 0; i < p.getInventory().getContainerSize(); i++) {
            ItemStack s = p.getInventory().getItem(i);
            if (s.is(BuiltInRegistries.ITEM.get(ResourceLocation.parse("bettergold:raw_flamegold")))) {
                landed = s;
            }
        }
        return "：RecipeManager=" + managerResult + " / slotsChanged后结果格=" + r1.getCount() + "x"
                + BuiltInRegistries.ITEM.getKey(r1.getItem()) + " / 补格=" + refill
                + " / 补格后(点击前)结果格=" + result.getCount() + "x"
                + " / 落进背包=" + landed.getCount() + "x"
                + BuiltInRegistries.ITEM.getKey(landed.getItem()) + " freeSlot=" + free
                + " / 网格读回=" + grid.toString().trim() + " / 菜单craftSlots查配方=" + menuQuery
                + " / doLimitedCrafting=" + limited;
    }

    // ---- C：金玫瑰丛 × 金染土 ----
    private static void partC() {
        BlockState dirt = AllBlocks.GOLD_INFUSED_DIRT.get().defaultBlockState();
        BlockState farmland = AllBlocks.GOLD_INFUSED_FARMLAND.get().defaultBlockState();
        ItemStack rose = new ItemStack(AllBlocks.GOLDEN_ROSE_BUSH_ITEM.get());

        readout("C0_sustain_readout", "READOUT :: 金玫瑰丛下半块存活链（TriState）：金染土="
                + sustain(dirt, "bettergold:golden_rose_bush")
                + " / 金染耕地=" + sustain(farmland, "bettergold:golden_rose_bush")
                + " / 石头（= 改前金染土的等价判据形状：裸 Block + 不在 #minecraft:dirt + 非 FarmBlock）="
                + sustain(Blocks.STONE.defaultBlockState(), "bettergold:golden_rose_bush"));

        // ① 金染土（作者的诉求）
        BlockPlaceResult r1 = tryPlace(100, dirt, rose);
        check("C1_rose_on_gold_dirt", r1.placed && r1.lowerOk && r1.upperOk,
                "真右键放到**金染土**上 ⇒ " + r1.describe());

        // ② 金染耕地（回归）
        BlockPlaceResult r2 = tryPlace(101, farmland, rose);
        check("C2_rose_on_gold_farmland", r2.placed && r2.lowerOk && r2.upperOk,
                "真右键放到**金染耕地**上 ⇒ " + r2.describe());

        // ③ 普通泥土 / 草方块（原版基线：**能**放）/ 石头（不能放）
        BlockPlaceResult r3 = tryPlace(102, Blocks.DIRT.defaultBlockState(), rose);
        check("C3a_rose_on_plain_dirt_vanilla", r3.placed,
                "普通**泥土** ⇒ " + r3.describe() + "（原版玫瑰丛本来就能放 ⇒ 这是**基线**，不是要求「不能放」）");
        BlockPlaceResult r4 = tryPlace(103, Blocks.GRASS_BLOCK.defaultBlockState(), rose);
        check("C3b_rose_on_grass_vanilla", r4.placed,
                "**草方块** ⇒ " + r4.describe() + "（同上，原版基线）");
        BlockPlaceResult r5 = tryPlace(104, Blocks.STONE.defaultBlockState(), rose);
        check("C3c_rose_on_stone_negative", !r5.placed,
                "**石头** ⇒ " + r5.describe() + "（负向对照：非土方块仍然放不上）");

        // ⑤ 金染土**不许**变成通用土：普通作物（小麦种子）不能直接种在金染土上
        BlockPlaceResult w1 = tryPlace(105, dirt, new ItemStack(Items.WHEAT_SEEDS));
        check("C4_wheat_not_on_gold_dirt", !w1.placed,
                "小麦种子种在**金染土**上 ⇒ " + w1.describe()
                        + "（我把白名单收成「只有金玫瑰丛」就是为了守住这一条："
                        + "否则 CropBlock#canSurvive 会先看 canSustainPlant ⇒ 作物绕过「必须先耕成金染耕地」）");
        BlockPlaceResult w2 = tryPlace(106, farmland, new ItemStack(Items.WHEAT_SEEDS));
        check("C5_wheat_on_gold_farmland", w2.placed,
                "小麦种子种在**金染耕地**上 ⇒ " + w2.describe() + "（回归）");
    }

    private static String sustain(BlockState soil, String plantId) {
        BlockPos pos = new BlockPos(400, 199, 400);
        BlockState plant = BuiltInRegistries.BLOCK.get(ResourceLocation.parse(plantId)).defaultBlockState();
        TriState t = soil.canSustainPlant(level, pos, Direction.UP, plant);
        return String.valueOf(t);
    }

    private record BlockPlaceResult(boolean placed, boolean lowerOk, boolean upperOk, String got) {
        String describe() {
            return "放上了=" + placed + " / 格子状态=" + got + " / 下半=lower:" + lowerOk + " 上半=upper:" + upperOk;
        }
    }

    /** 用**真右键路径**（`ServerPlayerGameMode#useItemOn`）把物品放到指定土块上。 */
    private static BlockPlaceResult tryPlace(int index, BlockState soil, ItemStack item) {
        BlockPos soilPos = new BlockPos(400, 200, 400 + index * 4);
        BlockPos target = soilPos.above();
        level.setBlockAndUpdate(soilPos, soil);
        level.setBlockAndUpdate(target, Blocks.AIR.defaultBlockState());
        level.setBlockAndUpdate(target.above(), Blocks.AIR.defaultBlockState());

        ServerPlayer p = newPlayer("c-" + index);
        p.setPos(soilPos.getX() + 0.5, soilPos.getY() + 1.0, soilPos.getZ() + 0.5);
        p.setItemInHand(InteractionHand.MAIN_HAND, item.copy());
        BlockHitResult hit = new BlockHitResult(
                new Vec3(soilPos.getX() + 0.5, soilPos.getY() + 1.0, soilPos.getZ() + 0.5),
                Direction.UP, soilPos, false);
        InteractionResult res;
        try {
            res = p.gameMode.useItemOn(p, level, p.getMainHandItem(), InteractionHand.MAIN_HAND, hit);
        } catch (Throwable t) {
            return new BlockPlaceResult(false, false, false, "<useItemOn 抛异常:" + t + ">");
        }
        BlockState got = level.getBlockState(target);
        boolean placed = !got.isAir();
        boolean lowerOk = false;
        boolean upperOk = false;
        if (got.hasProperty(BlockStateProperties.DOUBLE_BLOCK_HALF)) {
            lowerOk = got.getValue(BlockStateProperties.DOUBLE_BLOCK_HALF) == DoubleBlockHalf.LOWER;
            BlockState up = level.getBlockState(target.above());
            upperOk = up.getBlock() == got.getBlock()
                    && up.hasProperty(BlockStateProperties.DOUBLE_BLOCK_HALF)
                    && up.getValue(BlockStateProperties.DOUBLE_BLOCK_HALF) == DoubleBlockHalf.UPPER;
        }
        return new BlockPlaceResult(placed, lowerOk, upperOk,
                BuiltInRegistries.BLOCK.getKey(got.getBlock()) + " res=" + res);
    }
