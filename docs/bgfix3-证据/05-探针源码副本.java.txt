/**
 * bg-fix3 A 级实机探针（**只读验证，不参与任何玩法**；收尾整块删除）。
 *
 * <p>会话隔离：整包 {@code com.hjmmd_8.bettergold.probe.bgfix3}，收尾只删这个子目录。</p>
 * <p>开关文件：{@code <gameDir>/bgfix3-probe.enabled}（按 {@link FMLPaths#GAMEDIR} 算，
 * 不是按工程根算 —— 踩过的坑见 mcmod_experience {@code ex/03} §3.2 规则 3）。</p>
 * <p>世界守卫：第 1 tick 反射读 {@code getLevelName()}，**必须**是 {@code bgfix3probe}；
 * 不等 ⇒ 打 {@code ABORT} 并停止一切动作（**不调 halt** —— 万一真进了作者的存档，
 * halt 会走正常存盘把它重写一遍，那正是红线 5 的事故形态）。</p>
 *
 * <h2>本轮要证的四件事</h2>
 * <ol>
 *   <li>①/② 闪耀藤条的**掉落判据**：8 族 × 11 类物品 × {树叶, 藤蔓} 的真实 GLM 路径读数
 *       （{@code Block.getDrops(...)} ⇒ {@code LootTable#getRandomItems} ⇒
 *       {@code CommonHooks.modifyLoot} ⇒ 本模组的 GLM），外加命中概率对时运的曲线、
 *       以及在**真世界**里 {@code destroyBlock} 后数 {@code ItemEntity} 的交叉印证。</li>
 *   <li>③ 乐事小刀：真 {@code ServerPlayer} 拿到 {@code <族>_knife} ⇒
 *       「获得任意一种 XX金武器工具」**真的解锁**（改前对照那一轮应红）。</li>
 *   <li>⑤ 根成就：真玩家**获得手册物品** ⇒ root 解锁；{@code recipe_crafted} 那条老路径不再解锁。</li>
 *   <li>⑥ 八族盔甲：加载后的树是四级链、{@code frame == CHALLENGE}、集齐 4 件才解锁。</li>
 * </ol>
 *
 * <p>⚠ **成就一律用真 {@code ServerPlayer}**（{@code PlayerAdvancements#award} 第一句就是
 * {@code instanceof FakePlayer ⇒ return false}）。探针里的 {@link ProbePlayer} 是
 * {@code ServerPlayer} 的子类、**从不加进世界、从不 tick**，只用来驱动判据：这样既拿到
 * "真玩家"的授予链，又不需要哑连接（{@code connection} 只在 {@code ServerPlayer#tick →
 * flushDirty} 与容器同步器里被用到；本探针不 tick，并把 {@code broadcastChanges} 包在
 * try/catch 里单独计数）。</p>
 */

    // ==================== 读数小工具 ====================

    private static String id(String raw) {
        return raw.replace('/', '_');
    }

    private static void report(String rawId, boolean ok, String detail) {
        String key = id(rawId);
        if (ok) {
            pass++;
            bettergold.LOGGER.info("{} {} :: PASS {}", TAG, key, detail);
        } else {
            fail++;
            problems.add(key + " :: " + detail);
            bettergold.LOGGER.error("{} {} :: FAIL {}", TAG, key, detail);
        }
    }

    private static void skipped(String rawId, String detail) {
        skip++;
        bettergold.LOGGER.warn("{} {} :: SKIP {}", TAG, id(rawId), detail);
    }

    private static void readout(String rawId, String detail) {
        bettergold.LOGGER.info("{} {} :: READOUT {}", TAG, id(rawId), detail);
    }

    private static void abort(String why) {
        aborts.add(why);
        if (!aborted) {
            aborted = true;
            bettergold.LOGGER.error("{} ABORT {}（停止一切动作；**不调 halt**，避免把当前世界存盘）", TAG, why);
        }
    }

    // ==================== 主循环 ====================

    @SubscribeEvent
    public static void onServerTick(ServerTickEvent.Post event) {
        if (!armed) {
            Path flag = FMLPaths.GAMEDIR.get().resolve("bgfix3-probe.enabled");
            if (!Files.isRegularFile(flag)) {
                return;
            }
            armed = true;
            server = event.getServer();
            level = server.overworld();
            bettergold.LOGGER.info("{} armed=true gamedir={} flag={}", TAG, FMLPaths.GAMEDIR.get(), flag);
        }
        if (aborted) {
            return;
        }
        tick++;
        try {
            dispatch();
        } catch (Throwable t) {
            fail++;
            problems.add("阶段 tick=" + tick + " 抛异常：" + t);
            bettergold.LOGGER.error("{} 阶段 tick={} 抛异常", TAG, tick, t);
        }
    }

    private static void dispatch() {
        if (tick == T_GUARD) {
            guardWorld();
        } else if (tick == T_FORCELOAD) {
            level.setChunkForced(0, 0, true);
            readout("forced_chunks", "forced=" + level.getForcedChunks().size());
        } else if (tick == T_SETUP) {
            setupPlatform();
        } else if (tick == T_DROP_MATRIX) {
            runDropMatrix();
        } else if (tick == T_REAL_BREAK_0) {
            breakCase(0);
        } else if (tick == T_REAL_ASSERT_0) {
            assertCase(0);
        } else if (tick == T_REAL_BREAK_1) {
            breakCase(1);
        } else if (tick == T_REAL_ASSERT_1) {
            assertCase(1);
        } else if (tick == T_REAL_BREAK_2) {
            breakCase(2);
        } else if (tick == T_REAL_ASSERT_2) {
            assertCase(2);
        } else if (tick == T_ADV_META) {
            advanceMeta();
        } else if (tick == T_ADV_KNIFE) {
            advanceKnife();
        } else if (tick == T_ADV_ROOT) {
            advanceRoot();
        } else if (tick == T_ADV_ARMOR) {
            advanceArmor();
        } else if (tick == T_ADV_NEGATIVE) {
            advanceNegative();
        } else if (tick == T_SUMMARY) {
            summary();
        }
    }

    // ==================== 世界守卫（红线 5）====================

    private static void guardWorld() {
        Object data = level.getLevelData();
        try {
            Method m = data.getClass().getMethod("getLevelName");
            levelName = String.valueOf(m.invoke(data));
        } catch (Throwable t) {
            abort("probe_world_guard 读不到世界名：" + t.getClass().getSimpleName() + " " + t.getMessage());
            return;
        }
        boolean ok = PROBE_WORLD.equals(levelName);
        report("probe_world_guard", ok,
                "levelName=" + levelName + " expect=" + PROBE_WORLD + " authorWorld=" + AUTHOR_WORLD);
        if (levelName.equals(AUTHOR_WORLD)) {
            abort("世界名 = 作者存档名（红线 5）");
        } else if (!ok) {
            abort("世界名不是本轮自己的 ASCII 探针世界");
        }
    }

    private static void setupPlatform() {
        for (int dx = -2; dx <= 2; dx++) {
            for (int dz = -2; dz <= 2; dz++) {
                for (int dy = 0; dy <= 2; dy++) {
                    level.setBlockAndUpdate(TEST_POS.offset(dx, dy, dz), Blocks.AIR.defaultBlockState());
                }
            }
        }
        level.setBlockAndUpdate(TEST_POS.below(), Blocks.STONE.defaultBlockState());
        if (player == null) {
            GameProfile profile = new GameProfile(
                    UUID.nameUUIDFromBytes("bgfix3:probe".getBytes(StandardCharsets.UTF_8)), "bgfix3-probe");
            player = new ProbePlayer(server, level, profile);
            readout("probe_player", "class=" + player.getClass().getName()
                    + " isFakePlayer=" + player.isFakePlayer()
                    + "（真 ServerPlayer 子类；从不加进世界、从不 tick）");
            // 哑连接：**必须**给 connection（`award → rewards.grant → awardRecipesByKey →
            // ServerRecipeBook#sendRecipes → connection.send`）——否则**别的**带 `rewards.recipes`
            // 的成就在发奖励那一步 NPE，会把 vanilla 的 listener 循环打断，
            // 于是我们那一条判据根本没轮到就被跳过了（本轮第一版实测：8 条武器成就全红）。
            try {
                player.connection = new ServerGamePacketListenerImpl(
                        server, new DummyConnection(), player, CommonListenerCookie.createInitial(profile, false));
                readout("probe_connection", "dummy connection 已装上：" + player.connection.getClass().getName()
                        + "（send 全部 no-op）");
            } catch (Throwable t) {
                skipped("probe_connection", "哑连接装不上（" + t + "）⇒ 带 rewards 的成就会在发奖励那步抛异常");
            }
        }
        readout("setup_platform", "pos=" + TEST_POS + " air5x3x5 + stone below");
    }

    // ==================== ①/② 掉落判据 ====================

    private static boolean hasVine(List<ItemStack> drops) {
        Item vine = MetalSpecialItems.GLITTERING_VINE.get();
        for (ItemStack s : drops) {
            if (s.is(vine)) {
                return true;
            }
        }
        return false;
    }

    /** 真实 GLM 路径取 N 次掉落（`Block.getDrops` → `LootTable#getRandomItems` → `CommonHooks.modifyLoot`）。 */
    private static int rolls(BlockState state, ItemStack toolStack, int n) {
        int hits = 0;
        for (int i = 0; i < n; i++) {
            if (hasVine(Block.getDrops(state, level, TEST_POS, null, null, toolStack))) {
                hits++;
            }
        }
        return hits;
    }

    private static ItemStack tool(String itemId) {
        Item item = BuiltInRegistries.ITEM.get(ResourceLocation.parse(itemId));
        return item == Items.AIR ? ItemStack.EMPTY : new ItemStack(item);
    }

    private static void runDropMatrix() {
        BlockState leaves = Blocks.OAK_LEAVES.defaultBlockState();
        BlockState vine = Blocks.VINE.defaultBlockState();
        BlockState stone = Blocks.STONE.defaultBlockState();
        int n = 400;

        readout("drop_matrix_head",
                "path=Block.getDrops(state, level, pos, null, null, tool) -> LootTable#getRandomItems"
                        + " -> CommonHooks.modifyLoot -> AddGlitteringVineModifier; rolls/case=" + n);

        for (String metal : METALS) {
            for (String kind : KINDS) {
                String itemId = "bettergold:" + metal + "_" + kind;
                ItemStack t = tool(itemId);
                if (t.isEmpty()) {
                    skipped("drop_matrix_" + metal + "_" + kind, "物品不存在（可选模组没装？）：" + itemId);
                    continue;
                }
                int hLeaves = rolls(leaves, t, n);
                int hVine = rolls(vine, t, n);
                boolean expect = !"sturdygold".equals(metal) && !NOT_A_WEAPON.contains(kind);
                matrix.put(metal + "/" + kind + "/leaves", hLeaves);
                matrix.put(metal + "/" + kind + "/vine", hVine);
                report("drop_matrix_" + metal + "_" + kind,
                        (hLeaves > 0) == expect && (hVine > 0) == expect,
                        itemId + " leaves=" + hLeaves + "/" + n + " vine=" + hVine + "/" + n
                                + " expectDrop=" + expect);
            }
        }

        int iron = rolls(leaves, new ItemStack(Items.IRON_SWORD), n);
        int dia = rolls(leaves, new ItemStack(Items.DIAMOND_PICKAXE), n);
        int hand = rolls(leaves, ItemStack.EMPTY, n);
        report("drop_control_iron_sword", iron == 0, "iron_sword leaves=" + iron + "/" + n + " expect=0");
        report("drop_control_diamond_pickaxe", dia == 0, "diamond_pickaxe leaves=" + dia + "/" + n + " expect=0");
        report("drop_control_empty_hand", hand == 0, "empty-hand leaves=" + hand + "/" + n + " expect=0");
        ItemStack thornsPick = tool("bettergold:thornsgold_pickaxe");
        if (!thornsPick.isEmpty()) {
            int onStone = rolls(stone, thornsPick, n);
            report("drop_control_non_target_block", onStone == 0,
                    "thornsgold_pickaxe stone=" + onStone + "/" + n + " expect=0（不是树叶/藤蔓）");
        }

        ItemStack fortuneSword = tool("bettergold:thornsgold_sword");
        if (!fortuneSword.isEmpty()) {
            Optional<Holder.Reference<Enchantment>> fortune =
                    level.registryAccess().registryOrThrow(Registries.ENCHANTMENT).getHolder(Enchantments.FORTUNE);
            int[] hits = new int[4];
            int fn = 2000;
            for (int lvl = 0; lvl <= 3; lvl++) {
                ItemStack t = fortuneSword.copy();
                if (lvl > 0) {
                    if (fortune.isEmpty()) {
                        skipped("drop_fortune_l" + lvl, "注册表里没有 minecraft:fortune");
                        continue;
                    }
                    t.enchant(fortune.get(), lvl);
                }
                hits[lvl] = rolls(leaves, t, fn);
                double p = 0.06 + 0.06 * lvl;
                double sigma = Math.sqrt(fn * p * (1.0 - p));
                double dev = Math.abs(hits[lvl] - fn * p);
                report("drop_fortune_l" + lvl, dev <= 4.0 * sigma,
                        "fortune=" + lvl + " p=" + p + " expect=" + (fn * p) + " hits=" + hits[lvl]
                                + "/" + fn + " dev=" + String.format(java.util.Locale.ROOT, "%.1f", dev)
                                + " 4sigma=" + String.format(java.util.Locale.ROOT, "%.1f", 4.0 * sigma));
            }
            report("drop_fortune_monotonic",
                    hits[0] < hits[1] && hits[1] < hits[2] && hits[2] < hits[3],
                    "hits=" + hits[0] + "," + hits[1] + "," + hits[2] + "," + hits[3] + "（应严格递增）");
        }
    }

    // ---- 真世界 destroyBlock + 数 ItemEntity（交叉印证"真的掉进世界"）----

    private static final int REAL_N = 200;
    private static final String[] REAL_CASES = {"thornsgold_sword", "sturdygold_sword", "iron"};

    private static void clearItemEntities() {
        for (ItemEntity ie : level.getEntitiesOfClass(ItemEntity.class, new AABB(TEST_POS).inflate(16))) {
            ie.discard();
        }
    }

    private static void breakCase(int index) {
        clearItemEntities();
        String which = REAL_CASES[index];
        ItemStack t = which.equals("iron") ? new ItemStack(Items.IRON_SWORD) : tool("bettergold:" + which);
        if (t.isEmpty()) {
            skipped("drop_real_" + which, "工具不存在");
            return;
        }
        player.setItemInHand(InteractionHand.MAIN_HAND, t);
        BlockState leaves = Blocks.OAK_LEAVES.defaultBlockState();
        int ok = 0;
        for (int i = 0; i < REAL_N; i++) {
            level.setBlockAndUpdate(TEST_POS, leaves);
            // ⚠ 必须走 `ServerPlayerGameMode#destroyBlock`：`Level#destroyBlock` 传的是
            //   `ItemStack.EMPTY` 当工具（源码实测，见 docs/1.6-规格.md §二十二），
            //   拿它测"某个工具挖树叶掉不掉藤条"会得到**恒 0** 的假读数（本轮第一版就是这么错的）。
            if (player.gameMode.destroyBlock(TEST_POS)) {
                ok++;
            }
        }
        readout("drop_real_" + which + "_breaks",
                "gameMode.destroyBlock ok=" + ok + "/" + REAL_N + " tool=" + t.getItem()
                        + " mainhand=" + player.getMainHandItem().getItem());
    }

    private static void assertCase(int index) {
        String which = REAL_CASES[index];
        ItemStack t = which.equals("iron") ? new ItemStack(Items.IRON_SWORD) : tool("bettergold:" + which);
        if (t.isEmpty()) {
            return;
        }
        Item vine = MetalSpecialItems.GLITTERING_VINE.get();
        int vineHits = 0;
        int other = 0;
        for (ItemEntity ie : level.getEntitiesOfClass(ItemEntity.class, new AABB(TEST_POS).inflate(16))) {
            if (ie.getItem().is(vine)) {
                vineHits++;
            } else {
                other++;
            }
        }
        boolean expect = which.equals("thornsgold_sword");
        report("drop_real_" + which,
                (vineHits > 0) == expect,
                "真世界 ItemEntity：vine=" + vineHits + " other(树叶自身掉落)=" + other
                        + "（" + REAL_N + " 次真 destroyBlock；expectVine=" + expect + "）");
        clearItemEntities();
        level.setBlockAndUpdate(TEST_POS, Blocks.AIR.defaultBlockState());
    }

    // ==================== 成就侧（真 ServerPlayer）====================

    private static AdvancementHolder advHolder(String path) {
        return server.getAdvancements().get(ResourceLocation.parse("bettergold:" + path));
    }

    private static boolean done(String path) {
        AdvancementHolder h = advHolder(path);
        if (h == null) {
            problems.add("成就不存在（没能加载）：bettergold:" + path);
            return false;
        }
        return player.getAdvancements().getOrStartProgress(h).isDone();
    }

    private static void revokeAll(String path) {
        AdvancementHolder h = advHolder(path);
        if (h == null) {
            return;
        }
        for (String c : new ArrayList<>(h.value().criteria().keySet())) {
            player.getAdvancements().revoke(h, c);
        }
    }

    /** 真实获得一件物品：进背包 →（试）菜单广播 → 真 {@code INVENTORY_CHANGED} 触发器。
     *
     * <p>⚠⚠ **必须用 {@code stack.copy()} 去喂触发器**（本轮实测踩到的坑）：
     * {@code Inventory#add(ItemStack)} 会把**传入的那一份**拆进背包槽位、**把它清空**；
     * 如果拿同一个对象去调触发器，判据拿到的是**空栈** ⇒
     * {@code InventoryChangeTrigger} 的 `!stack.isEmpty() && items.get(0).test(stack)` 恒假 ⇒
     * **"获得即解锁"永远不成立**（而且一条日志都没有，看着像"标签没生效"）。
     * 症状极有迷惑性：背包**满了**之后 `add()` 加不进去、那一份反而非空 ⇒ 只有最后几个用例变绿。</p>
     *
     * <p>⚠ 判据的"授予"这一步与"奖励发放"这一步是**两段**（{@code PlayerAdvancements#award}：
     * {@code grantProgress} → {@code unregisterListeners} → {@code rewards().grant(player)}）：
     * 本探针的哑玩家没有 {@code connection}，而带 `rewards.recipes` 的成就会在**第二段**里
     * 走 {@code ServerRecipeBook#sendRecipes → connection.send} ⇒ NPE（本轮已用哑连接消除）。</p>
     */
    private static void obtain(ItemStack stack) {
        ItemStack changed = stack.copy();
        player.getInventory().add(stack); // 会消耗掉传入的这一份，所以触发器那份要提前 copy
        try {
            player.inventoryMenu.broadcastChanges();
        } catch (Throwable t) {
            broadcastFailures++;
        }
        try {
            CriteriaTriggers.INVENTORY_CHANGED.trigger(player, player.getInventory(), changed);
        } catch (Throwable t) {
            rewardNpe++;
            lastRewardNpe = t.getClass().getSimpleName() + ": " + t.getMessage();
        }

    /**

     * 证明"授予链本身通不通"，从而把红/绿定位到"派发"还是"授予"。

    private static void diagDirectAward() {
        AdvancementHolder h = advHolder("metal/flamegold/weapon");
        if (h == null) {
            return;
        }
        String[] crits = h.value().criteria().keySet().toArray(new String[0]);
        String target = crits.length > 1 ? crits[1] : crits[0];
        try {
            boolean awarded = player.getAdvancements().award(h, target);
            readout("diag_direct_award", "award(" + target + ")=" + awarded
                    + " done=" + done("metal/flamegold/weapon")
                    + "（真授予链；用来把『派发』与『授予』分开定位）");
        } catch (Throwable t) {
            readout("diag_direct_award", "award(" + target + ") 抛异常：" + t);
        }
        revokeAll("metal/flamegold/weapon");
        // 再直接问一次"这一格能不能通过判据"（把 HolderSet 拿出来自己 test 一遍）

            Object itemsObj = ti.getClass().getMethod("items").invoke(ti);
            ItemStack knife = tool("bettergold:flamegold_knife");

        } catch (Throwable t) {
            readout("diag_predicate_test", "反射失败：" + t);
        }
    }

    /**
     * 诊断读数（bg-fix3 第一轮 A 级踩到的坑）：把"判据里那个 HolderSet 到底解析成了什么"
     * 与"触发器上到底登记了几个 listener"都打出来。
     *
     * <p>{@code Criterion} 是 record：判据实例在 {@code triggerInstance()} 上，
     * 用 {@code trigger()} 会拿到触发器本身（mcmod_experience {@code ex/04} 第 67 条）。</p>
     */
    private static void diagPredicates() {
        AdvancementHolder h = advHolder("metal/flamegold/weapon");
        if (h == null) {
            return;
        }
        Item knife = BuiltInRegistries.ITEM.get(ResourceLocation.parse("bettergold:flamegold_knife"));
        for (var e : h.value().criteria().entrySet()) {
            Object ti = e.getValue().triggerInstance();
            StringBuilder sb = new StringBuilder();
            try {
                Object itemsObj = ti.getClass().getMethod("items").invoke(ti);
                for (Object p : (List<?>) itemsObj) {
                    Optional<HolderSet<Item>> set = ((ItemPredicate) p).items();
                    int size = set.map(HolderSet::size).orElse(-1);
                    boolean hasKnife = knife != Items.AIR
                            && set.map(s -> s.contains(knife.builtInRegistryHolder())).orElse(false);
                    sb.append("[size=").append(size).append(" hasKnife=").append(hasKnife).append("] ");
                }
            } catch (Throwable t) {
                sb.append("<反射失败 ").append(t).append('>');
            }
            readout("diag_predicate_" + e.getKey(),
                    "triggerInstance=" + ti.getClass().getSimpleName() + " items=" + sb);
        }
        try {
            java.lang.reflect.Field f = null;            for (Class<?> c = CriteriaTriggers.INVENTORY_CHANGED.getClass(); c != null && f == null;
                 c = c.getSuperclass()) {
                try {
                    f = c.getDeclaredField("players");
                } catch (NoSuchFieldException ignored) {
                    // 往上找
                }
            }
            if (f == null) {
                readout("diag_listeners", "找不到 players 字段");
            } else {
                f.setAccessible(true);
                Object map = f.get(CriteriaTriggers.INVENTORY_CHANGED);
                Object set = ((Map<?, ?>) map).get(player.getAdvancements());
                readout("diag_listeners",
                        "INVENTORY_CHANGED listeners(player)=" + (set == null ? "null" : ((Collection<?>) set).size())
                                + " playersEntries=" + ((Map<?, ?>) map).size());
                if (set != null) {
                    // 逐条打印"哪个成就的哪条判据"登记在触发器上（决定性读数）
                    for (Object listener : (Collection<?>) set) {
                        Object adv = listener.getClass().getMethod("advancement").invoke(listener);
                        Object crit = listener.getClass().getMethod("criterion").invoke(listener);
                        String advId = String.valueOf(adv.getClass().getMethod("id").invoke(adv));
                        if (advId.startsWith("bettergold:metal/flamegold") || advId.equals("bettergold:root")) {
                            readout("diag_listener_entry", advId + "#" + crit);
                        }
                    }
                }
            }
        } catch (Throwable t) {
            readout("diag_listeners", "反射失败：" + t);
        }
    }

    private static void advanceKnife() {
        for (String metal : METALS) {
            String adv = "metal/" + metal + "/weapon";
            revokeAll(adv);
            ItemStack knife = tool("bettergold:" + metal + "_knife");
            if (knife.isEmpty()) {
                skipped("adv_knife_" + metal, "刀不存在（没装乐事）：bettergold:" + metal + "_knife");
                continue;
            }
            if (metal.equals("flamegold")) {
                readout("diag_knife_criterion", "before trigger: " + criterionState(adv));
                diagListenerFor(adv, knife);
            }
            obtain(knife);
            if (metal.equals("flamegold")) {
                readout("diag_knife_criterion", "after trigger: " + criterionState(adv));
                diagListenerFor(adv, knife);
            }
            report("adv_knife_" + metal, done(adv),
                    "获得 bettergold:" + metal + "_knife -> " + adv + " done=" + done(adv));
        }
    }

    /** 逐条判据的达成状态（用来把"派发没跑"与"requirements 没自洽"分开定位）。 */
    private static String criterionState(String path) {
        AdvancementHolder h = advHolder(path);
        if (h == null) {
            return "<no advancement>";
        }
        var prog = player.getAdvancements().getOrStartProgress(h);
        StringBuilder sb = new StringBuilder("isDone=").append(prog.isDone()).append(' ');
        for (String c : h.value().criteria().keySet()) {
            var cp = prog.getCriterion(c);
            sb.append(c).append('=').append(cp == null ? "null" : cp.isDone()).append(' ');
        }
        return sb.toString();
    }

    /**
     * 反射读"触发器上到底登记了谁的 listener"，并对**那一个 listener 自己的** TriggerInstance
     * 手工调一次 {@code matches(...)} —— 把"没登记"与"登记了但 test 为假"彻底分开。
     */
    private static void diagListenerFor(String path, ItemStack stack) {
        try {
            java.lang.reflect.Field f = null;
            for (Class<?> c = CriteriaTriggers.INVENTORY_CHANGED.getClass(); c != null && f == null;
                 c = c.getSuperclass()) {
                try {
                    f = c.getDeclaredField("players");
                } catch (NoSuchFieldException ignored) {
                    // 往上找
                }
            }
            if (f == null) {
                readout("diag_listener_for", "找不到 players 字段");
                return;
            }
            f.setAccessible(true);
            Object setRaw = ((Map<?, ?>) f.get(CriteriaTriggers.INVENTORY_CHANGED))
                    .get(player.getAdvancements());
            if (setRaw == null) {
                readout("diag_listener_for", path + "：触发器上没有本玩家的 listener 集合");
                return;
            }
            Collection<?> set = (Collection<?>) setRaw;
            String want = "bettergold:" + path;
            int found = 0;
            for (Object listener : set) {
                Object adv = listener.getClass().getMethod("advancement").invoke(listener);
                String advId = String.valueOf(adv.getClass().getMethod("id").invoke(adv));
                if (!advId.equals(want)) {
                    continue;
                }
                found++;
                Object crit = listener.getClass().getMethod("criterion").invoke(listener);
                Object ti = listener.getClass().getMethod("trigger").invoke(listener);
                Object matched = ti.getClass()
                        .getMethod("matches", net.minecraft.world.entity.player.Inventory.class,
                                ItemStack.class, int.class, int.class, int.class)
                        .invoke(ti, player.getInventory(), stack, 0, 36, 0);
                readout("diag_listener_for", want + "#" + crit + " 自己的 TriggerInstance.matches=" + matched
                        + "（手工调用：inventory/stack/0/36/0）");
            }
            if (found == 0) {
                readout("diag_listener_for", want + " **没有**登记在 INVENTORY_CHANGED 上（found=0）");
            }
        } catch (Throwable t) {
            readout("diag_listener_for", "反射失败：" + t);
        }
    }
