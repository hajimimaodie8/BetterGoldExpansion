package com.hjmmd_8.bettergold.probe;

import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.UUID;

import com.google.gson.Gson;
import com.google.gson.GsonBuilder;
import com.mojang.authlib.GameProfile;
import com.mojang.datafixers.util.Pair;

import net.minecraft.advancements.AdvancementHolder;
import net.minecraft.advancements.AdvancementProgress;
import net.minecraft.advancements.CriteriaTriggers;
import net.minecraft.core.BlockPos;
import net.minecraft.core.HolderSet;
import net.minecraft.core.registries.BuiltInRegistries;
import net.minecraft.core.registries.Registries;
import net.minecraft.network.Connection;
import net.minecraft.network.protocol.Packet;
import net.minecraft.network.protocol.PacketFlow;
import net.minecraft.resources.ResourceKey;
import net.minecraft.resources.ResourceLocation;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.server.network.CommonListenerCookie;
import net.minecraft.server.network.ServerGamePacketListenerImpl;
import net.minecraft.world.item.Item;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.level.Level;
import net.minecraft.world.level.levelgen.structure.Structure;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

/**
 * 1.7 大轮 C 段的 A 级探针：真 {@code ServerPlayer} 解锁新增成就。
 *
 * <p>5 条走**自然触发**（`Inventory#add` → 容器监听 → `minecraft:inventory_changed`）；
 * 「红中黄室」走**真 `minecraft:location` 判据**（在下界找一座金之牢狱、传送进去、触发 LOCATION）。
 * 另外验证被毙掉的两条成就已不在产物 / 注册表里。</p>
 *
 * <p>开关文件 {@code run/bgbig-c-probe.enabled}；{@code halt(false)} 一次；forceload 双向归零。</p>
 */
@EventBusSubscriber(modid = "bettergold")
public final class CProbe {

    private static final String TAG = "[bgbig-c2]";
    private static final Path OUT = Path.of("probe", "bgbig");
    private static final Path SWITCH = Path.of("bgbig-c-probe.enabled");
    private static final Gson GSON = new GsonBuilder().setPrettyPrinting().disableHtmlEscaping().create();

    private static final int TICK_WAIT = 25;
    private static int passes;
    private static int fails;
    private static int phase;
    private static int tick;
    private static boolean halted;
    private static boolean done;
    private static ServerLevel level;
    private static ServerPlayer player;
    private static final List<Map<String, Object>> ROWS = new ArrayList<>();

    private CProbe() {
    }

    private static final Logger LOG = LoggerFactory.getLogger("CProbe");

    @SubscribeEvent
    public static void onServerStarted(ServerStartedEvent event) {
        if (!Files.isRegularFile(SWITCH)) {
            LOG.info("{} switch absent -> skipped", TAG);
            return;
        }
        if (done) {
            return;
        }
        done = true;
        try {
            setup(event.getServer());
            phase = 1;
            tick = 0;
        } catch (Throwable t) {
            LOG.error("{} setup failed", TAG, t);
            row("probe_setup", false, String.valueOf(t));
            finish(event.getServer());
        }
    }

    @SubscribeEvent
    public static void onTick(ServerTickEvent.Post event) {
        if (phase == 0 || halted) {
            return;
        }
        if (++tick < TICK_WAIT) {
            return;
        }
        try {
            run(event.getServer());
        } catch (Throwable t) {
            LOG.error("{} run failed", TAG, t);
            row("probe_run", false, String.valueOf(t));
        }
        finish(event.getServer());
    }

    private static void row(String name, boolean ok, String detail) {
        Map<String, Object> m = new LinkedHashMap<>();
        m.put("name", name);
        m.put("ok", ok);
        m.put("detail", detail);
        ROWS.add(m);
        if (ok) {
            passes++;
            LOG.info("{} PASS {} :: {}", TAG, name, detail);
        } else {
            fails++;
            LOG.error("{} FAIL {} :: {}", TAG, name, detail);
        }
    }

    private static void finish(MinecraftServer server) {
        if (halted) {
            return;
        }
        halted = true;
        try {
            Map<String, Object> root = new LinkedHashMap<>();
            root.put("probe", "bgbig-c-v1");
            root.put("pass", passes);
            root.put("fail", fails);
            root.put("rows", ROWS);
            Files.createDirectories(OUT);
            Files.writeString(OUT.resolve("c-report.json"), GSON.toJson(root), StandardCharsets.UTF_8);
            LOG.info("{} SUMMARY pass={} fail={}", TAG, passes, fails);
        } catch (Exception e) {
            LOG.error("{} report write failed", TAG, e);
        } finally {
            LOG.info("{} halting server (once)", TAG);
            server.halt(false);
        }
    }

    private static void setup(MinecraftServer server) {
        // 「红中黄室」要在**下界**里站进金之牢狱 ⇒ 玩家一开始就建在下界（避免跨维度 teleport 碰连接层）
        level = server.getLevel(Level.NETHER);
        GameProfile profile = new GameProfile(
                UUID.nameUUIDFromBytes("bgbig:c-probe".getBytes(StandardCharsets.UTF_8)), "bgbig-c-probe");
        player = new ProbePlayer(server, level, profile);
        player.moveTo(8.5D, 100.0D, 8.5D, 0.0F, 0.0F);
        try {
            player.connection = new ServerGamePacketListenerImpl(
                    server, new DummyConnection(), player, CommonListenerCookie.createInitial(profile, false));
        } catch (Throwable t) {
            LOG.warn("{} dummy connection failed: {}", TAG, String.valueOf(t));
        }
        // ⚠ `inventory_changed` 的容器监听只在 `initInventoryMenu()` 里挂（不是构造器）；
        //   而且它内部会 `setSynchronizer` → 发初始数据 ⇒ **必须在 connection 装好之后**再调。
        player.initInventoryMenu();
        row("probe_player", !player.isFakePlayer() && player.getAdvancements() != null,
                "class=" + player.getClass().getName() + " isFakePlayer=" + player.isFakePlayer()
                        + " dimension=" + player.level().dimension().location());
    }

    private static AdvancementHolder holder(MinecraftServer server, String id) {
        return server.getAdvancements().get(ResourceLocation.parse(id));
    }

    private static boolean done(MinecraftServer server, String id) {
        AdvancementHolder h = holder(server, id);
        if (h == null) {
            return false;
        }
        AdvancementProgress p = player.getAdvancements().getOrStartProgress(h);
        return p.isDone();
    }

    private static void give(String id) {
        Item item = BuiltInRegistries.ITEM.get(ResourceLocation.parse(id));
        if (item == null || item == net.minecraft.world.item.Items.AIR) {
            LOG.warn("{} item missing: {}", TAG, id);
            return;
        }
        player.getInventory().add(new ItemStack(item));
    }

    private static void run(MinecraftServer server) {
        // ---- 5 条走自然 inventory_changed ----
        give("bettergold:chaos_coin_string");
        give("bettergold:chaosgold_enchanting_book");
        give("minecraft:copper_block");
        give("bettergold:golden_torchflower_seeds");
        give("bettergold:golden_torchflower");
        give("bettergold:refined_alchemic_oil");
        // ⚠ `inventory_changed` 的派发点在 `ServerPlayer` 的**容器监听**里，由
        //   `inventoryMenu.broadcastChanges()` 驱动；探针玩家不 tick ⇒ 手动广播一次。
        player.inventoryMenu.broadcastChanges();
        row("award_dance", done(server, "bettergold:metal/chaosgold/dance"),
                "与金共舞：获得混沌金币串");
        row("award_enchant", done(server, "bettergold:metal/chaosgold/enchant"),
                "赋予无序之咒：获得混沌金附魔书");
        row("award_craft", done(server, "bettergold:metal/chaosgold/craft"),
                "金工细作：获得铜块（含混沌金币配方产物之一）");
        row("award_torchflower", done(server, "bettergold:agriculture/golden_torchflower"),
                "黄金火焰！：种子 + 花（组间 AND）");
        row("award_refined_oil", done(server, "bettergold:agriculture/refined_oil"),
                "膏状的油将升华为精炼的油：获得高精度炼金油");

        // ---- 红中黄室：真 location 判据 + 在下界找一座金之牢狱 ----
        boolean located = false;
        String locateDetail = "未尝试";
        try {
            ServerLevel nether = server.getLevel(Level.NETHER);
            var reg = nether.registryAccess().registryOrThrow(Registries.STRUCTURE);
            ResourceKey<Structure> key = ResourceKey.create(Registries.STRUCTURE,
                    ResourceLocation.parse("bettergold:golden_prison"));
            var holder = reg.getHolderOrThrow(key);
            HolderSet<Structure> set = HolderSet.direct(holder);
            Pair<BlockPos, ?> found = nether.getChunkSource().getGenerator().findNearestMapStructure(
                    nether, set, new BlockPos(0, 64, 0), 64, false);
            if (found == null) {
                locateDetail = "64 区块内没有金之牢狱";
            } else {
                BlockPos pos = found.getFirst();
                nether.setChunkForced(pos.getX() >> 4, pos.getZ() >> 4, true);
                // locate 给的是"结构起点"；`LocationPredicate` 要的是**某一块结构拼图内**的坐标
                // ⇒ 用同一个 `getStructureWithPieceAt` 在 y 轴上找第一处命中（保证判据能过）。
                BlockPos safe = pos;
                for (int y = -64; y <= 160; y++) {
                    BlockPos cand = new BlockPos(pos.getX(), y, pos.getZ());
                    if (nether.structureManager().getStructureWithPieceAt(cand, holder.value()).isValid()) {
                        safe = cand;
                        break;
                    }
                }
                // 同维度移动：`teleportTo(ServerLevel,...)` 会走连接层（哑连接没 channel）⇒ 用 setPos
                player.setPos(safe.getX() + 0.5D, safe.getY() + 1.0D, safe.getZ() + 0.5D);
                CriteriaTriggers.LOCATION.trigger(player);
                nether.setChunkForced(pos.getX() >> 4, pos.getZ() >> 4, false);
                located = true;
                locateDetail = "found=" + pos + " piece=" + safe
                        + " dimension=" + player.level().dimension().location();
            }
        } catch (Throwable t) {
            locateDetail = "locate failed: " + t;
        }
        row("award_golden_prison", done(server, "bettergold:treasure/golden_prison"),
                "红中黄室：进入金之牢狱（location+structures）:: " + locateDetail);

        // ---- 被毙掉的两条：产物 + 注册表都必须没有 ----
        row("removed_chaos_coin", holder(server, "bettergold:treasure/chaos_coin") == null,
                "「获得混沌金币」成就已毙（注册表查不到）");
        row("removed_chaosalloy_supply", holder(server, "bettergold:metal/chaosalloy/supply") == null,
                "「崩乱合金模板」成就已毙（注册表查不到）");

        // ---- 树挂点：以混沌金锭为基 ----
        AdvancementHolder alloy = holder(server, "bettergold:metal/chaosalloy/ingot");
        row("chaosalloy_parent_is_chaosgold_ingot",
                alloy != null && alloy.value().parent().isPresent()
                        && alloy.value().parent().get().toString()
                                .equals("bettergold:metal/chaosgold/ingot"),
                "崩乱合金锭的 parent = " + (alloy == null ? "?" : alloy.value().parent()));
        long ours = server.getAdvancements().getAllAdvancements().stream()
                .filter(h -> h.id().getNamespace().equals("bettergold")).count();
        row("adv_count_63", ours == 63, "本模组注册表成就数 = " + ours
                + "（总注册表含原版：" + server.getAdvancements().getAllAdvancements().size() + "）");
    }

    static final class ProbePlayer extends ServerPlayer {
        ProbePlayer(MinecraftServer srv, ServerLevel lvl, GameProfile profile) {
            super(srv, lvl, profile, net.minecraft.server.level.ClientInformation.createDefault());
        }
    }

    static final class DummyConnection extends Connection {
        DummyConnection() {
            super(PacketFlow.SERVERBOUND);
        }

        @Override
        public void send(Packet<?> packet) {
        }

        @Override
        public void send(Packet<?> packet, net.minecraft.network.PacketSendListener listener) {
        }

        @Override
        public void send(Packet<?> packet, net.minecraft.network.PacketSendListener listener, boolean flush) {
        }

        @Override
        public boolean isMemoryConnection() {
            return false;
        }
    }
}
