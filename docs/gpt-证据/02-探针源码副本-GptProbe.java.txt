    /** 最近一次伤害读数（key = 实体 id） */
    private static final Map<Integer, Float> PRE = new LinkedHashMap<>();
    private static final Map<Integer, Float> INCOMING = new LinkedHashMap<>();
    private static final Map<Integer, String> PRE_TYPE = new LinkedHashMap<>();
    private static final Map<Integer, String> PRE_ENTITY = new LinkedHashMap<>();
    private static final List<String> SOUNDS = new ArrayList<>();

    private GptProbe() {

    // ==================== 读数捕获（@SubscribeEvent，静态注册） ====================

    /** 护甲前（原始值） */
    @SubscribeEvent(priority = EventPriority.HIGHEST)
    public static void onIncoming(LivingIncomingDamageEvent event) {
        if (phase > 0 && !event.getEntity().level().isClientSide()) {
            INCOMING.put(event.getEntity().getId(), event.getAmount());
        }
    }

    /** 全部减免之后（LOWEST ⇒ 跑在本模组的 NORMAL 处理器之后） */
    @SubscribeEvent(priority = EventPriority.LOWEST)
    public static void onPre(LivingDamageEvent.Pre event) {
        if (phase > 0 && !event.getEntity().level().isClientSide()) {
            int id = event.getEntity().getId();
            PRE.put(id, event.getNewDamage());
            PRE_TYPE.put(id, event.getSource().typeHolder().unwrapKey()
                    .map(k -> k.location().toString()).orElse("?"));
            Entity owner = event.getSource().getEntity();
            PRE_ENTITY.put(id, owner == null ? "<无>" :
                    BuiltInRegistries.ENTITY_TYPE.getKey(owner.getType()) + "/" + owner.getUUID());
        }
    }

    /** 服务端音效（Level#playSound 的 NeoForge 事件） */
    @SubscribeEvent
    public static void onSound(PlayLevelSoundEvent.AtPosition event) {
        if (phase > 0) {
            var sound = event.getSound();
            SOUNDS.add((sound == null ? "?" : sound.unwrapKey()
                    .map(k -> k.location().toString()).orElse("?")) + "@" + event.getPosition());
        }
    }

    // ==================== 生命周期 ====================

    @SubscribeEvent
    public static void onServerStarted(ServerStartedEvent event) {
        if (!Files.isRegularFile(SWITCH)) {
            LOG.info("{} switch {} absent -> skipped", TAG, SWITCH.toAbsolutePath());
            return;
        }
        if (done) {
            return;
        }
        done = true;
        try {
            setup(event.getServer());
            phase = 1;
            tick = 0;
        } catch (Throwable t) {
            LOG.error("{} setup failed", TAG, t);
            row("probe_setup", false, String.valueOf(t));
            finish(event.getServer());
        }
    }

    @SubscribeEvent
    public static void onTick(ServerTickEvent.Post event) {
        if (phase == 0 || halted) {
            return;
        }
        if (++tick < TICK_WAIT) {
            return;
        }
        tick = 0;
        if (phase == 1) {
            phase = 2;
            try {
                runMain(event.getServer());
            } catch (Throwable t) {
                LOG.error("{} main phase failed", TAG, t);
                row("probe_main", false, String.valueOf(t));
            }
            return;
        }
        try {
            runLate(event.getServer());
        } catch (Throwable t) {
            LOG.error("{} late phase failed", TAG, t);
            row("probe_late", false, String.valueOf(t));
        }
        finish(event.getServer());
    }

    private static void finish(MinecraftServer server) {
        if (halted) {
            return;
        }
        halted = true;
        try {
            Map<String, Object> root = new LinkedHashMap<>();
            root.put("probe", "gpt-v1");
            root.put("pass", passes);
            root.put("fail", fails);
            root.put("forcedNow", level == null ? -1 : level.getForcedChunks().size());
            root.put("rows", ROWS);
            root.put("sounds", SOUNDS);
            Files.createDirectories(OUT);
            Files.writeString(OUT.resolve("report.json"), GSON.toJson(root), StandardCharsets.UTF_8);
            LOG.info("{} SUMMARY pass={} fail={} forcedNow={}", TAG, passes, fails,
                    level == null ? -1 : level.getForcedChunks().size());
        } catch (Exception e) {
            LOG.error("{} report write failed", TAG, e);
        } finally {
            LOG.info("{} halting server (once)", TAG);
            server.halt(false);
        }
    }

    // ==================== 工具 ====================

    private static void row(String name, boolean ok, String detail) {
        Map<String, Object> m = new LinkedHashMap<>();
        m.put("name", name);
        m.put("ok", ok);
        m.put("detail", detail);
        ROWS.add(m);
        if (ok) {
            passes++;
            LOG.info("{} PASS {} :: {}", TAG, name, detail);
        } else {
            fails++;
            LOG.error("{} FAIL {} :: {}", TAG, name, detail);
        }
    }

    private static void check(String name, boolean ok, String detail) {
        row(name, ok, detail);
    }

    private static Item item(String id) {
        return BuiltInRegistries.ITEM.get(ResourceLocation.parse(id));
    }

    private static ItemStack stack(String id, int n) {

    }

    private static String idOf(ItemStack s) {
        return s.isEmpty() ? "<空>" : BuiltInRegistries.ITEM.getKey(s.getItem()).toString();
    }

    }

    // ---------- ① 金火把花：金染土可放 / 普通耕地与石头不可 / 金染耕地仍可 ----------
    private static void item1FlowerPlacement() {
        String flowerId = "bettergold:golden_torchflower";
        for (int i = 0; i < 4; i++) {
            BlockPos base = new BlockPos(900 + i, 120, 900);
            BlockPos above = base.above();
            level.setBlockAndUpdate(above, Blocks.AIR.defaultBlockState());
            dummy.setItemInHand(InteractionHand.MAIN_HAND, stack(flowerId, 1));
            BlockHitResult hit = new BlockHitResult(new Vec3(base.getX() + 0.5D, base.getY() + 1.0D,
                    base.getZ() + 0.5D), Direction.UP, base, false);
            UseOnContext ctx = new UseOnContext(level, dummy, InteractionHand.MAIN_HAND,
                    stack(flowerId, 1), hit);
            InteractionResult res = ((net.minecraft.world.item.BlockItem) item(flowerId)).useOn(ctx);
            String got = idOf(new ItemStack(level.getBlockState(above).getBlock().asItem()));
            String baseName = idOf(new ItemStack(level.getBlockState(base).getBlock().asItem()));
            // 0 金染土 / 1 金染耕地 / 2 普通耕地 ⇒ 可放（普通耕地是**原版**行为）；3 石头 ⇒ 不可
            boolean expect = i <= 2;
            check("gpt1_place_on_" + (expect ? "ok_" : "neg_") + baseName.replace(':', '_'),
                    (flowerId.equals(got)) == expect,
                    "底座=" + baseName + " 结果格=" + got + "（期望 " + (expect ? flowerId : "<空>")
                            + "）；useOn=" + res);
        }
        // 负向：不能在普通土上（不是 dirt 标签以外的特例）
        BlockPos base = new BlockPos(904, 120, 900);
        place(904, 120, 900, Blocks.DIRT.defaultBlockState());
        level.setBlockAndUpdate(base.above(), Blocks.AIR.defaultBlockState());
        BlockHitResult hit = new BlockHitResult(new Vec3(904.5D, 121.0D, 900.5D), Direction.UP, base, false);
        ((net.minecraft.world.item.BlockItem) item(flowerId)).useOn(new UseOnContext(level, dummy,
                InteractionHand.MAIN_HAND, stack(flowerId, 1), hit));
        check("gpt1_plain_dirt_vanilla_parity",
                idOf(new ItemStack(level.getBlockState(base.above()).getBlock().asItem())).equals(flowerId),
                "普通土上的结果=" + idOf(new ItemStack(level.getBlockState(base.above()).getBlock().asItem()))
                        + "（期望能放 = 原版 BushBlock 判据，未被本轮改动）");
    }

    // ---------- ② 混沌金：盾 426 / 韧性每件 2.0 / 击退每件 0.05 ----------
    private static void item2Chaosgold() {
        MetalFamily cg = MetalFamily.byId("chaosgold");
        MetalFamily sg = MetalFamily.byId("sturdygold");
        if (cg == null || sg == null) {
            check("gpt2_families", false, "找不到 chaosgold / sturdygold 家族");
            return;
        }
        check("gpt2_shield_durability", new ItemStack(cg.shield.get()).getMaxDamage() == 426,
                "chaosgold_shield getMaxDamage=" + new ItemStack(cg.shield.get()).getMaxDamage() + "（期望 426）");
        check("gpt2_sturdygold_shield_regression", new ItemStack(sg.shield.get()).getMaxDamage() == 3072,
                "sturdygold_shield getMaxDamage=" + new ItemStack(sg.shield.get()).getMaxDamage() + "（期望 3072，回归）");

        String[] names = {"helmet", "chestplate", "leggings", "boots"};
        Item[] pieces = {cg.helmet.get(), cg.chestplate.get(), cg.leggings.get(), cg.boots.get()};
        for (int i = 0; i < 4; i++) {
            String nm = "chaosgold_" + names[i];
            List<String> tough = mods(pieces[i], "minecraft:generic.armor_toughness");
            List<String> kb = mods(pieces[i], "minecraft:generic.knockback_resistance");
            check("gpt2_" + names[i] + "_toughness", tough.size() == 1 && tough.get(0).endsWith("=2.0000"),
                    nm + " ARMOR_TOUGHNESS=" + tough + "（期望 1 条 =2.0）");
            check("gpt2_" + names[i] + "_knockback", kb.size() == 1 && kb.get(0).endsWith("=0.0500"),
                    nm + " KNOCKBACK_RESISTANCE=" + kb + "（期望 1 条 =0.05）");
        }
        List<String> shieldKb = mods(cg.shield.get(), "minecraft:generic.knockback_resistance");
        check("gpt2_shield_knockback", shieldKb.size() == 2
                        && shieldKb.stream().allMatch(s -> s.endsWith("=0.0500")),
                "chaosgold_shield KNOCKBACK_RESISTANCE=" + shieldKb + "（期望主/副手各 1 条 =0.05）");
        // 负向对照：万坚金盔甲仍是韧性 6 / 击退 0.15，盾牌仍是 0.10
        List<String> sgTough = mods(sg.helmet.get(), "minecraft:generic.armor_toughness");
        List<String> sgKb = mods(sg.helmet.get(), "minecraft:generic.knockback_resistance");
        List<String> sgShieldKb = mods(sg.shield.get(), "minecraft:generic.knockback_resistance");
        check("gpt2_regression_other_families", sgTough.size() == 1 && sgTough.get(0).endsWith("=6.0000")
                        && sgKb.size() == 1 && sgKb.get(0).endsWith("=0.1500")
                        && sgShieldKb.size() == 2 && sgShieldKb.stream().allMatch(s -> s.endsWith("=0.1000")),
                "sturdygold 韧性=" + sgTough + " 击退=" + sgKb + " 盾=" + sgShieldKb
                        + "（期望 6.0 / 0.15 / 0.1×2，其它族不受影响）");
    }

    /** 读某物品默认属性表：返回 ["<属性id>=<数值>"] */
    private static List<String> mods(Item item, String attributeId) {
        List<String> out = new ArrayList<>();
        // ⚠ 1.21.1：`Item#getDefaultAttributeModifiers()` 无参版本是空实现（ArmorItem 只覆写了
        //   `getDefaultAttributeModifiers(ItemStack)`）⇒ 必须走 ItemStack 版本，否则恒返回空表。
        var attrs = new ItemStack(item).getAttributeModifiers();
        for (var entry : attrs.modifiers()) {
            String aid = entry.attribute().unwrapKey().map(k -> k.location().toString()).orElse("?");
            if (aid.equals(attributeId)) {
                // 4 位小数：`0.05F` 读成 double 是 0.05000000074505806，直接比字符串永远不中
                out.add(aid + "=" + String.format(java.util.Locale.ROOT, "%.4f",
                        entry.modifier().amount()));
            }
        }
        return out;
    }

    // ---------- ④ 易金商人新手：24 金蛋 → 1 礼品金票（真 MerchantOffer） ----------

        Villager v = EntityType.VILLAGER.create(level);
        if (v == null) {
            check("gpt4_villager_spawn", false, "create(VILLAGER) 返回 null");
            return;

        v.moveTo(1003.5D, 121.0D, 1000.5D, 0.0F, 0.0F);

                .setProfession(AllVillagers.GOLD_TRADER.get())

                .setType(net.minecraft.world.entity.npc.VillagerType.PLAINS));
        level.addFreshEntity(v);

        // 真实交易表：NeoForge 在 TagsUpdatedEvent 里把 VillagerTradesEvent 的结果写进
        // `VillagerTrades.TRADES`（VillagerTradingManager#postVillagerEvents）⇒ 这里读的就是**注册出来的那张表**。
        var table = net.minecraft.world.entity.npc.VillagerTrades.TRADES.get(AllVillagers.GOLD_TRADER.get());
        if (table == null) {
            check("gpt4_trade_table", false, "VillagerTrades.TRADES 里没有易金商人（本体事件没跑到）");
            return;
        }
        var random = level.getRandom();
        List<String> novice = new ArrayList<>();
        MerchantOffer hit = null;
        var noviceArr = table.get(1);
        if (noviceArr != null) {
            for (var listing : noviceArr) {
                MerchantOffer o = listing.getOffer(v, random);
                novice.add(idOf(o.getCostA()) + "×" + o.getCostA().getCount() + "→"
                        + idOf(o.getResult()) + "×" + o.getResult().getCount()
                        + " xp=" + o.getXp() + " uses=" + o.getMaxUses());
                if (o.getCostA().is(item("bettergold:golden_egg")) && o.getCostA().getCount() == 24
                        && o.getResult().is(item("bettergold:gift_gold_ticket"))) {
                    hit = o;
                }
            }
        }
        check("gpt4_offer_found", hit != null, "新手档真 MerchantOffer 列表：" + novice);
        if (hit != null) {
            check("gpt4_offer_shape", hit.getCostA().getCount() == 24 && hit.getResult().getCount() == 1
                            && hit.getXp() == 10 && hit.getMaxUses() == 999,
                    "成本=" + idOf(hit.getCostA()) + "×" + hit.getCostA().getCount()
                            + " 产物=" + idOf(hit.getResult()) + "×" + hit.getResult().getCount()
                            + " xp=" + hit.getXp() + " maxUses=" + hit.getMaxUses()
                            + "（期望 24 / 1 / 10 / 999）");
        }
        // 新手档：2..5 级里不该有「24 金蛋」那条
        int higher = 0;
        for (int lvl = 2; lvl <= 5; lvl++) {
            var arr = table.get(lvl);
            if (arr == null) {
                continue;
            }
            for (var listing : arr) {
                MerchantOffer o = listing.getOffer(v, random);
                if (o.getCostA().is(item("bettergold:golden_egg")) && o.getCostA().getCount() == 24) {
                    higher++;
                }
            }
        }
        check("gpt4_novice_level_only", higher == 0,
                "2..5 级交易表里「24 金蛋」出现次数=" + higher + "（期望 0 ⇒ 只在新手档）");
        v.discard();
    }

        //   把效果时长设为 1，再调用一次 LivingEntity#tick 不会触发 Expired（在 MobEffectInstance#tick 里）
        //   ⇒ 这里显式用「提前 1 tick」的路径：removeEffect 不结算，所以只能自然到期。
        //   做法：记录读数占位，读到时差在 late phase 里复核。
        ZOMBIE_FOR_BURST = zombie;
        PRE_TYPE.remove(zombie.getId());
        PRE_ENTITY.remove(zombie.getId());
        PRE.remove(zombie.getId());

        // (b) 每件 25% 中毒伤害：0..4 件，真 neoforge:poison 伤害（用**真僵尸**当受害体：玩家实例
        //     不在世界里时有若干玩家专属早退，读数会全是 -1 —— 第一次跑就踩到了）
        MetalFamily vd = MetalFamily.byId("voodoogold");
        if (vd == null) {
            check("gpt6_voodoo_family", false, "找不到 voodoogold 家族");
            return;
        }
        var pv = EntityType.ZOMBIE.create(level);
        if (pv == null) {
            check("gpt6_poison_victim", false, "create(ZOMBIE) 返回 null");
            return;
        }
        pv.moveTo(970.5D, 121.0D, 975.5D, 0.0F, 0.0F);
        pv.setNoAi(true);
        level.addFreshEntity(pv);
        var dyn = level.registryAccess().registryOrThrow(Registries.DAMAGE_TYPE);
        var poison = dyn.getHolderOrThrow(NeoForgeMod.POISON_DAMAGE);
        DamageSource src = new DamageSource(poison);
        boolean bypassArmor = src.is(DamageTypeTags.BYPASSES_ARMOR);
        float raw = 8.0F;
        StringBuilder sb = new StringBuilder();
        float[] ratio = new float[5];
        for (int p = 0; p <= 4; p++) {
            for (EquipmentSlot slot : new EquipmentSlot[] {EquipmentSlot.HEAD, EquipmentSlot.CHEST,
                    EquipmentSlot.LEGS, EquipmentSlot.FEET}) {
                pv.setItemSlot(slot, ItemStack.EMPTY);
            }
            Item[] pieces = {vd.helmet.get(), vd.chestplate.get(), vd.leggings.get(), vd.boots.get()};
            EquipmentSlot[] slots = {EquipmentSlot.HEAD, EquipmentSlot.CHEST, EquipmentSlot.LEGS,
                    EquipmentSlot.FEET};
            for (int i = 0; i < p; i++) {
                pv.setItemSlot(slots[i], new ItemStack(pieces[i]));
            }
            pv.setHealth(pv.getMaxHealth());
            pv.invulnerableTime = 0;
            PRE.remove(pv.getId());
            INCOMING.remove(pv.getId());
            pv.hurt(src, raw);
            float pre = PRE.getOrDefault(pv.getId(), -1.0F);
            float inc = INCOMING.getOrDefault(pv.getId(), -1.0F);
            ratio[p] = inc > 0 ? pre / inc : -1.0F;
            sb.append("件=").append(p).append(" incoming=").append(inc).append(" pre=").append(pre)
                    .append(" 比值=").append(String.format(java.util.Locale.ROOT, "%.4f", ratio[p]))
                    .append("；");
        }
        pv.discard();
        POISON_STORY = "neoforge:poison bypasses_armor=" + bypassArmor + "；" + sb + "逐档比值=" 
                + String.format(java.util.Locale.ROOT, "%.4f/%.4f/%.4f/%.4f/%.4f",
                        ratio[0], ratio[1], ratio[2], ratio[3], ratio[4]);
        boolean ok = true;
        for (int p = 0; p <= 4; p++) {
            float want = 1.0F - 0.25F * p;
            float got = ratio[p];
            // 若中毒不吃护甲（bypasses_armor），比值必须**精确**等于 1-0.25p；
            // 若吃护甲，则护甲还乘一个因子 ⇒ 这里用「逐档等比」判据：相邻两档每次降 25% 的原始份额。
            boolean good = Math.abs(got - want) < 0.02F;
            if (!bypassArmor) {
                // 用第 0 档归一化：ratio[p] / ratio[0] 应 = 1-0.25p
                good = ratio[0] > 0 && Math.abs(got / ratio[0] - want) < 0.02F;
            }
            if (!good) {
                ok = false;
            }
        }
        check("gpt6_poison_per_piece", ok, POISON_STORY);
    }

    // ==================== 真合成（真 CraftingMenu + 两次真点击） ====================

    private static CraftingMenu craft(String tag, String[] grid, String wantId, int wantCount,
            boolean expectMatch) {
        BlockPos pos = new BlockPos(400, 200, 380);
        level.setBlockAndUpdate(pos.below(), Blocks.STONE.defaultBlockState());
        dummy.getInventory().clearContent();
        CraftingMenu menu = new CraftingMenu(0, dummy.getInventory(),
                ContainerLevelAccess.create(level, pos));
        dummy.containerMenu = menu;

        StringBuilder lc = new StringBuilder();
        int[] before = new int[9];
        for (int i = 0; i < 9; i++) {
            ItemStack s = grid[i] == null ? ItemStack.EMPTY : new ItemStack(item(grid[i]), 1);
            menu.getSlot(1 + i).set(s);
            before[i] = s.getCount();
            lc.append(grid[i] == null ? "-" : grid[i].substring(grid[i].indexOf(':') + 1)).append(' ');
        }
        menu.slotsChanged(menu.getSlot(1).container);
        LOG.info("{} INFO {} :: 摆位 {}", TAG, tag, lc.toString().trim());

        ItemStack result = menu.getSlot(0).getItem();
        boolean matched = !result.isEmpty();
        if (expectMatch) {
            check(tag + "_result", matched && idOf(result).equals(wantId) && result.getCount() == wantCount,
                    "结果格=" + (matched ? result.getCount() + "x " + idOf(result) : "<空>")
                            + "（期望 " + wantCount + "x " + wantId + "）");
        } else {
            check(tag + "_no_match", !matched, "负向：结果格=" + (matched ? idOf(result) : "<空>") + "（期望空）");
            return menu;
        }
        int free = 10;
        menu.clicked(0, 0, ClickType.PICKUP, dummy);
        boolean cleared = menu.getSlot(0).getItem().isEmpty();
        menu.clicked(free, 0, ClickType.PICKUP, dummy);

        List<String> wrong = new ArrayList<>();
        for (int i = 0; i < 9; i++) {
            int now = menu.getSlot(1 + i).getItem().getCount();
            int want = grid[i] == null ? 0 : Math.max(0, before[i] - 1);
            // 模具（金钱贝模具）走 `getRemainingItems`：以「耐久 −1、数量 1」回到同一格 ⇒ 放行
            if (grid[i] != null && grid[i].equals("bettergold:golden_cowrie_mold") && before[i] == 1) {
                want = 1;
            }
            if (now != want) {
                wrong.add("格" + i + "=" + now + "≠" + want);
            }
        }
        check(tag + "_consumed", wrong.isEmpty() && cleared,
                "逐格 −1 正确=" + wrong.isEmpty() + " 结果格已空=" + cleared + " 偏差=" + wrong);

        int landed = 0;
        for (int i = 0; i < dummy.getInventory().getContainerSize(); i++) {
            ItemStack s = dummy.getInventory().getItem(i);
            if (!s.isEmpty() && BuiltInRegistries.ITEM.getKey(s.getItem()).toString().equals(wantId)) {
                landed += s.getCount();
            }
        }
        check(tag + "_landed", landed == wantCount, "两次真点击后落进背包=" + landed + "x（期望 " + wantCount + "）");
        return menu;
    }
